# SSE events breakdown
with client.messages.stream(...) as stream:
    for event in stream:
        if event.type == "message_start":
            print(f"Message ID: {event.message.id}")
        elif event.type == "content_block_start":
            print(f"Block type: {event.content_block.type}")  # "text" | "thinking" | "tool_use"
        elif event.type == "content_block_delta":
            if event.delta.type == "text_delta":

            elif event.delta.type == "input_json_delta":
                print(event.delta.partial_json, end="")  # tool use args streaming
        elif event.type == "message_stop":
            print("\nDone!")

Full Tool Use Loop

import anthropic
import json

client = anthropic.Anthropic()

# Implement actual tool functions
def get_weather(location: str, unit: str = "celsius") -> dict:
    """Simulated weather API"""
    return {"location": location, "temperature": 22, "unit": unit,
            "conditions": "Sunny", "humidity": 65}

def search_database(query: str, category: str = "all", limit: int = 10) -> list:
    """Simulated database search"""
    return [{"id": 1, "name": f"{query} result", "category": category}]

TOOLS_MAP = {"get_weather": get_weather, "search_database": search_database}

def run_agent(user_message: str) -> str:
    messages = [{"role": "user", "content": user_message}]

    while True:
        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )

        # Add assistant's response to messages
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason == "end_turn":
            # Extract text response
            for block in response.content:
                if hasattr(block, "text"):
                    return block.text
            return "Done"

        elif response.stop_reason == "tool_use":
            # Process each tool call
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    tool_fn = TOOLS_MAP[block.name]
                    result = tool_fn(**block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(result)
                    })

            # Add tool results as user message
            messages.append({"role": "user", "content": tool_results})
        else:
            break

    return "Unexpected stop reason"

answer = run_agent("What's the weather in Paris and search for 'laptop' electronics?")
print(answer)

Parallel Tool Calls

import asyncio
import anthropic

async def run_tool_async(tool_name: str, tool_input: dict, tool_id: str) -> dict:
    """Execute a single tool asynchronously"""
    tool_fn = TOOLS_MAP[tool_name]
    result = await asyncio.to_thread(tool_fn, **tool_input)
    return {
        "type": "tool_result",
        "tool_use_id": tool_id,
        "content": json.dumps(result)
    }

async def run_agent_with_parallel_tools(user_message: str) -> str:
    client = anthropic.AsyncAnthropic()
    messages = [{"role": "user", "content": user_message}]

    while True:
        response = await client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason == "end_turn":
            return next((b.text for b in response.content if hasattr(b, "text")), "Done")

        elif response.stop_reason == "tool_use":
            tool_calls = [b for b in response.content if b.type == "tool_use"]

            # Execute ALL tool calls in PARALLEL
            tool_results = await asyncio.gather(*[
                run_tool_async(tc.name, tc.input, tc.id)
                for tc in tool_calls
            ])

            messages.append({"role": "user", "content": list(tool_results)})

# Claude may call get_weather(Tokyo) AND get_weather(London) AND search_database in one shot
# These execute concurrently — much faster than sequential
result = asyncio.run(run_agent_with_parallel_tools(
    "Compare weather in Tokyo, London, and Berlin, and search for weather apps"
))

Prompt Caching

Prompt Caching caches specific portions of your prompt on Anthropic's servers. Cache read discount: ~90% cheaper than full input pricing. Cache TTL: 5 minutes (default) or 1 hour (with extended cache). Minimum cacheable tokens: 1024 for Sonnet/Opus, 2048 for Haiku. 
import anthropic

client = anthropic.Anthropic()

# Large static content that you want to cache
large_document = "..." * 500  # A long document, codebase, or knowledge base

# Mark content with cache_control to cache it
response = client.messages.create(
    model="claude-3-5-sonnet-20241022",
    max_tokens=1024,
    system=[
        {
            "type": "text",
            "text": "You are a helpful assistant analyzing documents.",
        },
        {
            "type": "text",
            "text": large_document,
            "cache_control": {"type": "ephemeral"}  # Cache this block (5 min TTL)
        }
    ],
    messages=[{"role": "user", "content": "Summarize the document"}]
)

# Check caching stats in usage
print(response.usage.cache_creation_input_tokens)  # Tokens cached (first time)
print(response.usage.cache_read_input_tokens)       # Tokens read from cache (subsequent)
print(response.usage.input_tokens)                  # Non-cached tokens

Building Agents: Tool Loop Pattern

import anthropic
import json
from typing import Callable

class ClaudeAgent:
    def __init__(self, model: str = "claude-3-5-sonnet-20241022",
                 system: str = "You are a helpful assistant.",
                 tools: list = None):
        self.client = anthropic.Anthropic()
        self.model = model
        self.system = system
        self.tools = tools or []
        self.tool_implementations: dict[str, Callable] = {}

    def register_tool(self, name: str, fn: Callable):
        self.tool_implementations[name] = fn

    def run(self, user_message: str, conversation_history: list = None) -> tuple[str, list]:
        messages = (conversation_history or []) + [
            {"role": "user", "content": user_message}
        ]
        max_iterations = 20
        iteration = 0

        while iteration < max_iterations:
            iteration += 1

            try:
                response = self.client.messages.create(
                    model=self.model,
                    max_tokens=4096,
                    system=self.system,
                    tools=self.tools if self.tools else [],
                    messages=messages
                )
            except anthropic.RateLimitError:
                import time; time.sleep(60)
                continue
            except anthropic.APIStatusError as e:
                return f"API error: {e}", messages

            # Append assistant response
            messages.append({"role": "assistant", "content": response.content})

            if response.stop_reason == "end_turn":
                text = next((b.text for b in response.content
                             if hasattr(b, "text")), "")
                return text, messages

            elif response.stop_reason == "tool_use":
                tool_results = []
                for block in response.content:
                    if block.type != "tool_use":
                        continue
                    tool_name = block.name
                    if tool_name not in self.tool_implementations:
                        result = f"Error: Unknown tool '{tool_name}'"
                    else:
                        try:
                            result = self.tool_implementations[tool_name](**block.input)
                            result = json.dumps(result) if not isinstance(result, str) else result
                        except Exception as e:
                            result = f"Error executing {tool_name}: {str(e)}"

                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })

                messages.append({"role": "user", "content": tool_results})

            elif response.stop_reason == "max_tokens":
                return "Response truncated (max_tokens reached)", messages

        return "Max iterations reached", messages

Conversation History Management

from anthropic import Anthropic

client = Anthropic()

class ConversationManager:
    def __init__(self, max_tokens_in_context: int = 100000):
        self.messages = []
        self.max_tokens = max_tokens_in_context

    def estimate_tokens(self, messages: list) -> int:
        """Rough token estimate (actual counting requires tokenizer)"""
        total_chars = sum(
            len(str(m.get("content", ""))) for m in messages
        )
        return total_chars // 4  # ~4 chars per token

    def trim_history(self):
        """Keep messages within token budget"""
        while (len(self.messages) > 2 and
               self.estimate_tokens(self.messages) > self.max_tokens):
            # Remove oldest non-system pair (keep alternating structure)
            if len(self.messages) >= 2:
                self.messages.pop(0)  # Remove oldest user
                self.messages.pop(0)  # Remove oldest assistant

    def summarize_old_messages(self, keep_recent: int = 6):
        """Compress old history into a summary"""
        if len(self.messages) <= keep_recent:
            return
        old_messages = self.messages[:-keep_recent]
        summary_prompt = "Summarize this conversation in 3-4 sentences: " + \
            " | ".join(f"{m['role']}: {str(m['content'])[:200]}" for m in old_messages)

        summary_response = client.messages.create(
            model="claude-3-5-haiku-20241022",
            max_tokens=512,
            messages=[{"role": "user", "content": summary_prompt}]
        )
        summary = summary_response.content[0].text
        self.messages = [
            {"role": "user", "content": f"Previous conversation summary: {summary}"},
            {"role": "assistant", "content": "Understood. I'll keep that context in mind."}
        ] + self.messages[-keep_recent:]

    def chat(self, user_input: str, system: str = "") -> str:
        self.messages.append({"role": "user", "content": user_input})
        self.trim_history()

        response = client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=2048,
            system=system,
            messages=self.messages
        )
        assistant_reply = response.content[0].text
        self.messages.append({"role": "assistant", "content": assistant_reply})
        return assistant_reply

Q2: How does prompt caching work and when should you use it? Intermediate

Prompt caching stores specific prompt sections on Anthropic's infrastructure. When the same content is used again, you pay ~10% of the normal input price for cached tokens.

When to use:

  • Large static system prompts (>1024 tokens) reused across many requests
  • Long documents/codebases sent with multiple questions
  • Few-shot examples that don't change
  • Multi-turn conversations with the same large context

How: Add "cache_control": {"type": "ephemeral"} to the content block you want cached. The system matches on exact prefix — everything up to the cache_control marker is cached.

TTL: 5 minutes for ephemeral. The cache refreshes on each hit, so continuous use keeps it alive.

Q3: Explain the tool_use content block lifecycle in a multi-turn conversation. Intermediate

When Claude wants to use a tool:

  1. Claude returns stop_reason="tool_use" and content contains both text and tool_use blocks
  2. You add the assistant's full content (text + tool_use blocks) as the next "assistant" message
  3. You execute the tools, get results
  4. You add a "user" message with content = list of tool_result blocks (matching tool_use_id)
  5. Claude receives the results and continues generation

The conversation structure must alternate user/assistant strictly. Tool results are sent as a "user" message — this is why you must include both the assistant's tool_use blocks AND the tool results in the conversation history.

Q4: What is the difference between stop_reason "end_turn" and "tool_use"? Easy
  • end_turn: Claude finished naturally — final response is complete. Extract text content and return to user.
  • tool_use: Claude wants to call one or more tools. You must execute the tools and return results before Claude can continue. Do NOT show this response to the user.
  • max_tokens: Response was cut off. Consider increasing max_tokens or handling partial responses.
  • stop_sequence: Claude generated a custom stop sequence you specified in stop_sequences parameter.
Q5: How does extended thinking differ from standard Claude responses? Intermediate

Standard: Claude generates the response in one pass, mixing reasoning and output implicitly.

Extended thinking: Claude explicitly reasons in "thinking" blocks (visible as type="thinking" in content) before producing the final text. The thinking is more thorough, structured, and exploratory.

budget_tokens: Controls how many tokens Claude can use for thinking. Larger budget = more thorough reasoning, higher cost and latency. The total max_tokens must exceed budget_tokens.

Use cases: Multi-step math, complex coding problems, strategic planning, tasks where Claude needs to consider multiple approaches before committing.

Tradeoff: 2-3x latency and cost for complex tasks. For simple tasks, the thinking adds overhead without benefit.

Q6: How do you handle rate limits and retries in a production Claude application? Intermediate 
import anthropic
import time
import random

client = anthropic.Anthropic(max_retries=0)  # Disable SDK auto-retry, handle manually

def call_with_retry(fn, max_retries=5):
    for attempt in range(max_retries):
        try:
            return fn()
        except anthropic.RateLimitError as e:
            if attempt == max_retries - 1:
                raise
            # Exponential backoff with jitter
            wait = (2 ** attempt) + random.uniform(0, 1)
            print(f"Rate limited. Waiting {wait:.1f}s...")
            time.sleep(wait)
        except anthropic.APIConnectionError:
            if attempt == max_retries - 1:
                raise
            time.sleep(5)
        except anthropic.APIStatusError as e:
            if e.status_code == 529:  # Overloaded
                time.sleep(30)
            else:
                raise

Q7: What is MCP (Model Context Protocol) and why was it created? Intermediate

MCP is an open protocol by Anthropic for connecting AI models to data sources and tools in a standardized way.

Problem before MCP: Every AI application needed custom integrations for each tool (database connector, API client, file reader). N tools × M applications = N×M integrations to build and maintain.

MCP solution: Standardized client-server protocol. Build one MCP server for a tool, works with any MCP client (Claude Desktop, IDE plugins, custom apps).

Components:

  • MCP Server: exposes tools, resources, and prompts via the protocol
  • MCP Client: connects to servers (Claude Desktop, Cursor IDE)
  • MCP Host: application embedding the client

Q9: What are the best practices for structuring system prompts for agentic applications? Advanced

Effective agentic system prompt structure:

  1. Role and persona: "You are [role] with [expertise]. Your goal is [objective]."
  2. Capabilities: List available tools and when to use each
  3. Constraints: What Claude should NOT do ("Never delete files without confirmation")
  4. Output format: How to structure responses
  5. Error handling: "If uncertain, ask for clarification rather than proceeding"
  6. Context: Relevant domain knowledge as cached content

Cache large context: Put static knowledge base content with cache_control. Only the last few turns need to be fresh.

Q13: How do you implement a reliable extraction pipeline using Claude's structured output? Advanced 
from pydantic import BaseModel, Field
from typing import Optional, List
import anthropic
import json

class ExtractedInvoice(BaseModel):
    invoice_number: str
    vendor_name: str
    date: str  # YYYY-MM-DD
    total_amount: float
    line_items: List[dict]
    payment_terms: Optional[str]

# Method 1: Force tool use with the schema
def extract_invoice_via_tool(text: str) -> ExtractedInvoice:
    client = anthropic.Anthropic()
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        tools=[{
            "name": "extract_invoice",
            "description": "Extract invoice data from text",
            "input_schema": ExtractedInvoice.model_json_schema()
        }],
        tool_choice={"type": "tool", "name": "extract_invoice"},
        messages=[{"role": "user",
                   "content": f"Extract invoice data:\n\n{text}"}]
    )
    tool_use = next(b for b in response.content if b.type == "tool_use")
    return ExtractedInvoice(**tool_use.input)

# Method 2: Retry on validation failure
def extract_with_retry(text: str, max_retries: int = 3) -> ExtractedInvoice:
    errors = []
    for attempt in range(max_retries):
        try:
            result = extract_invoice_via_tool(text)
            return result
        except Exception as e:
            errors.append(str(e))
            # Tell Claude about the error on next attempt
            text = f"Previous attempt failed: {e}\nPlease fix and retry.\n\n{text}"
    raise ValueError(f"Extraction failed after {max_retries} attempts: {errors}")
Q14: What are the security considerations when building Claude-powered applications? Advanced
  • API key security: Never expose API key in frontend code. Always call Claude from server-side. Use environment variables.
  • Prompt injection: Sanitize user input before including in prompts. Use XML tags to clearly demarcate user content from instructions.
  • Jailbreak resistance: System prompt with clear prohibitions. Use metadata.user_id to help Anthropic's safety systems. Add content filtering layer.
  • Data privacy: Claude processes data you send — don't send PII unless necessary. Use prompt caching for system context (not user data).
  • Output validation: Never execute Claude's code output directly. Parse and validate structured outputs before using them.
  • Rate limiting per user: Implement your own rate limiting to prevent abuse of your Claude integration.
  • Audit logging: Log all inputs/outputs with user IDs for compliance and debugging.
