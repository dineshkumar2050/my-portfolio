2. Arrays

Two Pointers — Two Sum (sorted)

Easy 
function twoSumSorted(nums, target) {
  let l = 0, r = nums.length - 1;
  while (l < r) {
    const sum = nums[l] + nums[r];
    if (sum === target) return [l, r];
    sum < target ? l++ : r--;
  }
  return [];
}
// Time: O(n)  Space: O(1)

Sliding Window — Max Subarray Sum (size k)

Easy 
function maxSubarrayK(nums, k) {
  let sum = nums.slice(0, k).reduce((a, b) => a + b, 0);
  let max = sum;
  for (let i = k; i < nums.length; i++) {
    sum += nums[i] - nums[i - k];
    max = Math.max(max, sum);
  }
  return max;
}
// Kadane's for unbounded: O(n), O(1)

Prefix Sums — Trapping Rain Water

Hard 
function trap(height) {
  let l = 0, r = height.length - 1, lMax = 0, rMax = 0, water = 0;
  while (l < r) {
    if (height[l] <= height[r]) {
      lMax = Math.max(lMax, height[l]);
      water += lMax - height[l++];
    } else {
      rMax = Math.max(rMax, height[r]);
      water += rMax - height[r--];
    }
  }
  return water;
}
// Time: O(n)  Space: O(1)
Prefix sum array pre[i] = pre[i-1] + nums[i] enables range sum queries in O(1). Use it when asked about subarray sums.

3. Strings

Anagram Check

Easy 
function isAnagram(s, t) {
  if (s.length !== t.length) return false;
  const count = {};
  for (const c of s) count[c] = (count[c] || 0) + 1;
  for (const c of t) {
    if (!count[c]) return false;
    count[c]--;
  }
  return true;
}
// Time: O(n)  Space: O(1) — bounded by alphabet size

Longest Palindromic Substring (Expand Around Center)

Medium 
function longestPalindrome(s) {
  let res = '';
  const expand = (l, r) => {
    while (l >= 0 && r < s.length && s[l] === s[r]) { l--; r++; }
    return s.slice(l + 1, r);
  };
  for (let i = 0; i < s.length; i++) {
    const odd = expand(i, i), even = expand(i, i + 1);
    if (odd.length > res.length) res = odd;
    if (even.length > res.length) res = even;
  }
  return res;
}
// Time: O(n²)  Space: O(1)

Minimum Window Substring (Sliding Window)

Hard 
function minWindow(s, t) {
  const need = {}, have = {};
  for (const c of t) need[c] = (need[c] || 0) + 1;
  let formed = 0, required = Object.keys(need).length;
  let l = 0, min = Infinity, res = '';
  for (let r = 0; r < s.length; r++) {
    const c = s[r];
    have[c] = (have[c] || 0) + 1;
    if (need[c] && have[c] === need[c]) formed++;
    while (formed === required) {
      if (r - l + 1 < min) { min = r - l + 1; res = s.slice(l, r + 1); }
      have[s[l]]--;
      if (need[s[l]] && have[s[l]] < need[s[l]]) formed--;
      l++;
    }
  }
  return res;
}
// Time: O(n)  Space: O(k)
For substring problems, sliding window with a frequency map covers 90% of cases. Track "formed" (chars matching required count) to avoid re-scanning the map.

4. Linked Lists

Reverse a Linked List

Easy 
function reverseList(head) {
  let prev = null, curr = head;
  while (curr) {
    const next = curr.next;
    curr.next = prev;
    prev = curr;
    curr = next;
  }
  return prev;
}
// Time: O(n)  Space: O(1)

Detect Cycle — Floyd's Tortoise & Hare

Medium 
function hasCycle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next;
    fast = fast.next.next;
    if (slow === fast) return true;
  }
  return false;
}
// To find cycle start: reset slow to head, advance both by 1 until equal
// Time: O(n)  Space: O(1)

Merge Two Sorted Lists

Easy 
function mergeTwoLists(l1, l2) {
  const dummy = { next: null };
  let cur = dummy;
  while (l1 && l2) {
    if (l1.val <= l2.val) { cur.next = l1; l1 = l1.next; }
    else { cur.next = l2; l2 = l2.next; }
    cur = cur.next;
  }
  cur.next = l1 || l2;
  return dummy.next;
}
// Time: O(n+m)  Space: O(1)
Always use a dummy head node for linked list construction — it eliminates edge cases for empty lists and makes the code cleaner.

5. Stacks & Queues

Valid Parentheses (Stack)

Easy 
function isValid(s) {
  const stack = [], map = { ')': '(', ']': '[', '}': '{' };
  for (const c of s) {
    if ('([{'.includes(c)) stack.push(c);
    else if (stack.pop() !== map[c]) return false;
  }
  return stack.length === 0;
}
// Time: O(n)  Space: O(n)

Next Greater Element (Monotonic Stack)

Medium 
function nextGreaterElement(nums) {
  const res = new Array(nums.length).fill(-1);
  const stack = []; // stores indices, decreasing values
  for (let i = 0; i < nums.length; i++) {
    while (stack.length && nums[i] > nums[stack.at(-1)]) {
      res[stack.pop()] = nums[i];
    }
    stack.push(i);
  }
  return res;
}
// Monotonic stack: O(n) — each element pushed/popped once

Implement Queue with Two Stacks

Easy 
class MyQueue {
  constructor() { this.inbox = []; this.outbox = []; }
  push(x) { this.inbox.push(x); }
  pop() {
    if (!this.outbox.length)
      while (this.inbox.length) this.outbox.push(this.inbox.pop());
    return this.outbox.pop();
  }
  peek() { this.pop(); /* peek trick — add back */ }
  empty() { return !this.inbox.length && !this.outbox.length; }
}
// Amortized O(1) per operation
Monotonic stack pattern: maintain a stack of elements in increasing or decreasing order. Pop when the current element breaks the order — that's when you resolve the answer for popped elements.

BFS — Level Order Traversal

function levelOrder(root) {
  if (!root) return [];
  const res = [], queue = [root];
  while (queue.length) {
    const level = [];
    for (let i = queue.length; i > 0; i--) {
      const node = queue.shift();
      level.push(node.val);
      if (node.left) queue.push(node.left);
      if (node.right) queue.push(node.right);
    }
    res.push(level);
  }
  return res;
}
// Time: O(n)  Space: O(n)

DFS — Inorder / Preorder / Postorder

// Inorder: Left → Root → Right (sorted output for BST)
const inorder = (node, res = []) => {
  if (!node) return res;
  inorder(node.left, res); res.push(node.val); inorder(node.right, res);
  return res;
};
// Preorder: Root → Left → Right (copy tree, serialize)
const preorder = (node, res = []) => {
  if (!node) return res;
  res.push(node.val); preorder(node.left, res); preorder(node.right, res);
  return res;
};
// Postorder: Left → Right → Root (delete tree, evaluate expressions)
const postorder = (node, res = []) => {
  if (!node) return res;
  postorder(node.left, res); postorder(node.right, res); res.push(node.val);
  return res;
};

Max Depth & Path Sum

const maxDepth = (root) =>
  !root ? 0 : 1 + Math.max(maxDepth(root.left), maxDepth(root.right));

function hasPathSum(root, target) {
  if (!root) return false;
  if (!root.left && !root.right) return root.val === target;
  return hasPathSum(root.left, target - root.val) ||
         hasPathSum(root.right, target - root.val);
}
// Both O(n) time, O(h) space where h = tree height
BFS uses a queue (iterative), DFS uses recursion or an explicit stack. Use BFS for shortest path / level problems, DFS for path/subtree problems.

7. Binary Search Trees

Insert & Search

function insertBST(root, val) {
  if (!root) return { val, left: null, right: null };
  if (val < root.val) root.left = insertBST(root.left, val);
  else root.right = insertBST(root.right, val);
  return root;
}
const searchBST = (root, val) =>
  !root || root.val === val ? root :
  val < root.val ? searchBST(root.left, val) : searchBST(root.right, val);
// Average O(log n), worst O(n) for skewed tree

Validate BST

Medium 
function isValidBST(root, min = -Infinity, max = Infinity) {
  if (!root) return true;
  if (root.val <= min || root.val >= max) return false;
  return isValidBST(root.left, min, root.val) &&
         isValidBST(root.right, root.val, max);
}
// Time: O(n)  Space: O(h)

Lowest Common Ancestor

Medium 
function lowestCommonAncestor(root, p, q) {
  if (p.val < root.val && q.val < root.val)
    return lowestCommonAncestor(root.left, p, q);
  if (p.val > root.val && q.val > root.val)
    return lowestCommonAncestor(root.right, p, q);
  return root; // one on each side, or one equals root
}
// For general binary tree: check if p or q is in left/right subtrees
BST property: all left descendants < node < all right descendants. Inorder traversal of BST gives sorted array — classic interview trick to validate or process BST in sorted order.

8. Heaps / Priority Queues

JavaScript lacks a built-in heap. Simulate with a min-heap class or sort-based approach in interviews.

Min-Heap (Array-based)

class MinHeap {
  constructor() { this.h = []; }
  push(v) {
    this.h.push(v);
    let i = this.h.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (this.h[p] <= this.h[i]) break;
      [this.h[p], this.h[i]] = [this.h[i], this.h[p]]; i = p;
    }
  }
  pop() {
    const top = this.h[0];
    this.h[0] = this.h.pop();
    let i = 0;
    while (true) {
      let s = i, l = 2*i+1, r = 2*i+2;
      if (l < this.h.length && this.h[l] < this.h[s]) s = l;
      if (r < this.h.length && this.h[r] < this.h[s]) s = r;
      if (s === i) break;
      [this.h[s], this.h[i]] = [this.h[i], this.h[s]]; i = s;
    }
    return top;
  }
  peek() { return this.h[0]; }
  size() { return this.h.length; }
}

K Largest Elements

Medium 
function kLargest(nums, k) {
  // Use min-heap of size k — O(n log k)
  const heap = new MinHeap();
  for (const n of nums) {
    heap.push(n);
    if (heap.size() > k) heap.pop();
  }
  return heap.h; // k largest elements
}
// Alternatively: sort descending, take first k — O(n log n)
Key heap patterns: (1) K largest = min-heap of size K. (2) K smallest = max-heap of size K. (3) Median of stream = two heaps (max-heap for lower half, min-heap for upper half).

9. Hash Maps / Sets

Two Sum (Classic)

Easy 
function twoSum(nums, target) {
  const map = new Map(); // value -> index
  for (let i = 0; i < nums.length; i++) {
    const comp = target - nums[i];
    if (map.has(comp)) return [map.get(comp), i];
    map.set(nums[i], i);
  }
  return [];
}
// Time: O(n)  Space: O(n)

Group Anagrams

Medium 
function groupAnagrams(strs) {
  const map = new Map();
  for (const s of strs) {
    const key = s.split('').sort().join('');
    if (!map.has(key)) map.set(key, []);
    map.get(key).push(s);
  }
  return [...map.values()];
}
// Time: O(n * k log k)  Space: O(n*k)

LRU Cache

Hard 
class LRUCache {
  constructor(cap) { this.cap = cap; this.map = new Map(); }
  get(key) {
    if (!this.map.has(key)) return -1;
    const val = this.map.get(key);
    this.map.delete(key); this.map.set(key, val); // move to end
    return val;
  }
  put(key, val) {
    if (this.map.has(key)) this.map.delete(key);
    this.map.set(key, val);
    if (this.map.size > this.cap) this.map.delete(this.map.keys().next().value);
  }
}
// JS Map preserves insertion order — first key = LRU
// Time: O(1) get/put  Space: O(cap)
JS Map insertion order makes LRU cache elegant without a doubly-linked list. In other languages, combine HashMap + DoublyLinkedList for O(1) operations.

10. Graphs

BFS Template

function bfs(graph, start) {
  const visited = new Set([start]);
  const queue = [start];
  while (queue.length) {
    const node = queue.shift();
    for (const neighbor of graph[node] || []) {
      if (!visited.has(neighbor)) {
        visited.add(neighbor);
        queue.push(neighbor);
      }
    }
  }
}
// Time: O(V+E)  Space: O(V)

DFS Template

function dfs(graph, node, visited = new Set()) {
  if (visited.has(node)) return;
  visited.add(node);
  for (const neighbor of graph[node] || []) {
    dfs(graph, neighbor, visited);
  }
}
// Iterative DFS: replace recursion with stack

Topological Sort — Kahn's Algorithm (BFS)

Medium 
function topoSort(n, edges) {
  const indegree = new Array(n).fill(0), adj = Array.from({length:n}, ()=>[]);
  for (const [u, v] of edges) { adj[u].push(v); indegree[v]++; }
  const queue = [], res = [];
  for (let i = 0; i < n; i++) if (indegree[i] === 0) queue.push(i);
  while (queue.length) {
    const node = queue.shift(); res.push(node);
    for (const nb of adj[node]) if (--indegree[nb] === 0) queue.push(nb);
  }
  return res.length === n ? res : []; // [] = cycle detected
}
// Time: O(V+E)

Union-Find (with Path Compression)

class UnionFind {
  constructor(n) { this.parent = Array.from({length:n}, (_,i)=>i); this.rank=[...Array(n)].fill(0); }
  find(x) {
    if (this.parent[x] !== x) this.parent[x] = this.find(this.parent[x]);
    return this.parent[x];
  }
  union(x, y) {
    const [px, py] = [this.find(x), this.find(y)];
    if (px === py) return false;
    this.rank[px] >= this.rank[py] ? this.parent[py] = px : this.parent[px] = py;
    if (this.rank[px] === this.rank[py]) this.rank[px]++;
    return true;
  }
}
// Nearly O(1) per op with path compression + union by rank
Use BFS for shortest path (unweighted), DFS for connected components, Union-Find for dynamic connectivity, Kahn's for detecting cycles and ordering in DAGs.

11. Dynamic Programming

1D DP: Climbing Stairs

Easy 
function climbStairs(n) {
  if (n <= 2) return n;
  let [a, b] = [1, 2];
  for (let i = 3; i <= n; i++) [a, b] = [b, a + b];
  return b;
}
// Fibonacci variant. Time: O(n) Space: O(1)

1D DP: House Robber

Medium 
function rob(nums) {
  let [prev2, prev1] = [0, 0];
  for (const n of nums) {
    [prev2, prev1] = [prev1, Math.max(prev1, prev2 + n)];
  }
  return prev1;
}
// dp[i] = max(dp[i-1], dp[i-2] + nums[i])  Time: O(n) Space: O(1)

1D DP: Coin Change

Medium 
function coinChange(coins, amount) {
  const dp = new Array(amount + 1).fill(Infinity);
  dp[0] = 0;
  for (let i = 1; i <= amount; i++)
    for (const c of coins)
      if (c <= i) dp[i] = Math.min(dp[i], dp[i - c] + 1);
  return dp[amount] === Infinity ? -1 : dp[amount];
}
// Time: O(amount * coins)  Space: O(amount)

2D DP: Longest Common Subsequence

Medium 
function lcs(s, t) {
  const m = s.length, n = t.length;
  const dp = Array.from({length: m+1}, () => new Array(n+1).fill(0));
  for (let i = 1; i <= m; i++)
    for (let j = 1; j <= n; j++)
      dp[i][j] = s[i-1] === t[j-1] ? dp[i-1][j-1] + 1
               : Math.max(dp[i-1][j], dp[i][j-1]);
  return dp[m][n];
}
// Time: O(m*n)  Space: O(m*n), optimizable to O(n)

2D DP: Edit Distance

Hard 
function minDistance(w1, w2) {
  const m = w1.length, n = w2.length;
  const dp = Array.from({length:m+1}, (_,i) => Array.from({length:n+1},(_,j)=>i||j));
  for (let i = 1; i <= m; i++)
    for (let j = 1; j <= n; j++)
      dp[i][j] = w1[i-1] === w2[j-1] ? dp[i-1][j-1]
               : 1 + Math.min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1]);
  return dp[m][n];
}
// Time: O(m*n)  Space: O(m*n)

12. Binary Search

Search in Rotated Sorted Array

Medium 
function searchRotated(nums, target) {
  let l = 0, r = nums.length - 1;
  while (l <= r) {
    const m = (l + r) >> 1;
    if (nums[m] === target) return m;
    if (nums[l] <= nums[m]) { // left half sorted
      if (nums[l] <= target && target < nums[m]) r = m - 1;
      else l = m + 1;
    } else { // right half sorted
      if (nums[m] < target && target <= nums[r]) l = m + 1;
      else r = m - 1;
    }
  }
  return -1;
}
// Time: O(log n)

Find Peak Element

Medium 
function findPeakElement(nums) {
  let l = 0, r = nums.length - 1;
  while (l < r) {
    const m = (l + r) >> 1;
    if (nums[m] < nums[m + 1]) l = m + 1;
    else r = m;
  }
  return l;
}
// Binary search on answer: Time O(log n)

Search Insert Position

Easy 
function searchInsert(nums, target) {
  let l = 0, r = nums.length;
  while (l < r) {
    const m = (l + r) >> 1;
    if (nums[m] < target) l = m + 1;
    else r = m;
  }
  return l; // leftmost insertion point
}
// This is lower_bound pattern — returns first index >= target
Binary search template: use l < r for finding boundaries, l <= r for exact search. Always think about what happens at boundaries — off-by-one errors are the #1 binary search bug.

13. Sorting Algorithms

QuickSort

function quickSort(arr, l = 0, r = arr.length - 1) {
  if (l >= r) return;
  const pivot = arr[r];
  let i = l;
  for (let j = l; j < r; j++)
    if (arr[j] <= pivot) [arr[i], arr[j]] = [arr[j], arr[i++]];
  [arr[i], arr[r]] = [arr[r], arr[i]];
  quickSort(arr, l, i - 1);
  quickSort(arr, i + 1, r);
}
// Avg O(n log n), Worst O(n²) — unstable, in-place

MergeSort

function mergeSort(arr) {
  if (arr.length <= 1) return arr;
  const mid = arr.length >> 1;
  const [L, R] = [mergeSort(arr.slice(0, mid)), mergeSort(arr.slice(mid))];
  const res = [];
  let [i, j] = [0, 0];
  while (i < L.length && j < R.length)
    res.push(L[i] <= R[j] ? L[i++] : R[j++]);
  return [...res, ...L.slice(i), ...R.slice(j)];
}
// Always O(n log n) — stable, requires O(n) space

15. Interview Q&A

Easy

Easy What is the difference between O(n) and O(n²) in practice?
O(n) grows linearly — doubling input doubles time. O(n²) grows quadratically — doubling input quadruples time. For n=10,000: O(n) is 10k operations, O(n²) is 100 million operations. This is why nested loops over the same array are a red flag — always ask "can I precompute or use a hash map to reduce this to O(n)?"
Easy When should you use a hash map over an array?
Use a hash map when keys are non-sequential, sparse, or string/complex types. Use an array when keys are dense integers (0 to n). Hash maps give O(1) average for get/set but have overhead per entry (~50-100 bytes). Arrays give O(1) indexed access with minimal overhead. For counting character frequencies in a string, either works — a 26-element array is faster in practice.
Easy What does inorder traversal of a BST produce?
Inorder traversal (left → root → right) of a BST produces elements in sorted ascending order. This property is the key to many BST problems: to find the kth smallest element, just do inorder and return the kth element. To validate a BST, check that inorder output is strictly increasing.
Easy What is the space complexity of recursive DFS on a tree?
O(h) where h is the tree height. For a balanced tree, h = O(log n), so space is O(log n). For a skewed tree (essentially a linked list), h = O(n), so worst-case space is O(n). This call stack space is often overlooked — mention it explicitly in interviews to show thoroughness.

Intermediate

Medium Explain Floyd's cycle detection and how to find the cycle start.
Two pointers: slow moves 1 step, fast moves 2 steps. If there's a cycle, they meet inside it. To find the start: reset slow to head, keep fast at meeting point, advance both by 1 step at a time — they meet at the cycle start. Why? The distance from head to cycle start equals the distance from meeting point to cycle start (provable with modular arithmetic on cycle length). Time O(n), Space O(1).
Medium How does Kahn's topological sort detect cycles?
Kahn's starts with all nodes with in-degree 0 and processes them via BFS, decrementing neighbor in-degrees. If a cycle exists, the nodes in the cycle never reach in-degree 0, so they're never added to the queue. At the end, if the result array size is less than the total number of nodes, a cycle exists. This doubles as a cycle detector for directed graphs.
Medium When would you choose greedy over DP?
Greedy works when the greedy choice property holds: a locally optimal choice is always part of a globally optimal solution. This requires proof (exchange argument or induction). DP is safer — it considers all possibilities. Classic greedy problems: interval scheduling (sort by end time), Huffman encoding, jump game. If you're unsure, try DP first and optimize. Greedy usually runs in O(n log n) vs DP at O(n²) or O(n*k).
Medium Explain the monotonic stack pattern with an example.
A monotonic stack maintains elements in strictly increasing or decreasing order. When processing "Daily Temperatures": iterate left to right, push indices onto a decreasing stack. When nums[i] > nums[stack.top()], we found the "next warmer day" for the top element. Pop it and record the answer. Each element is pushed/popped at most once → O(n) total. The key insight: we defer answering until we find the trigger event (a larger/smaller element).
Medium What is the difference between BFS and Dijkstra's?
BFS finds shortest path in unweighted graphs (each edge has weight 1). It uses a queue and explores in order of hop count. Dijkstra's finds shortest path in weighted graphs (non-negative weights). It uses a min-heap (priority queue) and explores in order of cumulative distance. BFS is O(V+E); Dijkstra's is O((V+E) log V). For grids/mazes with equal-cost moves, BFS is sufficient and faster.

Advanced

Advanced How does path compression in Union-Find achieve near O(1) operations?
Path compression flattens the tree during find() by making every node on the path point directly to the root. Combined with union by rank (always attach smaller tree under larger tree root), the amortized cost per operation becomes O(α(n)) where α is the inverse Ackermann function — grows so slowly it's practically O(1) for any realistic n. Without path compression, find() is O(log n); without union by rank, it's O(n) in the worst case.
Advanced How would you implement a median-finding stream in O(log n) per insertion?
Use two heaps: a max-heap for the lower half and a min-heap for the upper half. Invariant: max-heap size equals min-heap size (even total) or is one larger (odd total). Insert: add to max-heap, then rebalance by moving top of max-heap to min-heap if max-heap top > min-heap top, then balance sizes. Median: if equal sizes, average both tops; else top of max-heap. Each insert is O(log n). getMedian is O(1).
Advanced Explain the time complexity of building a heap vs repeated insertions.
Repeated insertions (heapify up n times): O(n log n) total. Building a heap bottom-up (Floyd's heapify): O(n). Why? Bottom-up heapify processes nodes from bottom to top. Nodes at height h do O(h) work. Total work = Σ (n/2^h) * h ≈ n * Σ h/2^h = 2n → O(n). The key insight: most nodes (half the tree) are leaves doing O(1) work; only the O(log n) nodes near the root do O(log n) work.

Very Advanced

Very Advanced How does the edit distance DP state transition capture insert, delete, and replace?
dp[i][j] = min edits to convert word1[0..i-1] to word2[0..j-1]. If word1[i-1] == word2[j-1]: dp[i][j] = dp[i-1][j-1] (no operation needed). Else: dp[i][j] = 1 + min of: dp[i-1][j] (delete from word1 — we already converted word1[0..i-2] to word2[0..j-1], now delete word1[i-1]), dp[i][j-1] (insert into word1 — we converted word1[0..i-1] to word2[0..j-2], now insert word2[j-1]), dp[i-1][j-1] (replace — convert word1[0..i-2] to word2[0..j-2], replace word1[i-1] with word2[j-1]).
Very Advanced Explain why quicksort's worst case is O(n²) and how to mitigate it.
Worst case occurs when pivot selection produces maximally unbalanced partitions (one side has 0 elements, other has n-1). This happens with sorted/reverse-sorted input when always choosing first or last element as pivot. Mitigations: (1) Median-of-three: choose median of first, middle, last as pivot → avoids sorted-input worst case. (2) Random pivot: O(n²) worst case becomes astronomically unlikely in practice — expected O(n log n). (3) Introsort (used by C++ std::sort): falls back to heapsort if recursion depth exceeds O(log n), guaranteeing O(n log n) worst case.
Very Advanced How would you design an O(n) algorithm to find the kth largest element without sorting?
Quickselect (partition-based selection). Like quicksort but recurse only into the partition containing the kth element. After placing pivot at index p: if p == k-1, return pivot; if p > k-1, recurse left; else recurse right with adjusted k. Average case O(n) — each level roughly halves the problem: T(n) = T(n/2) + O(n) = O(n) by Master Theorem. Worst case O(n²) with bad pivot; use random pivot or median-of-medians (guaranteed O(n)) for adversarial inputs. Space: O(1) in-place.
Very Advanced Walk through how you would solve the "Trapping Rain Water" problem using multiple approaches and their trade-offs.
Approach 1 — Prefix/Suffix max arrays: Precompute leftMax[i] and rightMax[i]. Water at i = min(leftMax[i], rightMax[i]) - height[i]. O(n) time, O(n) space. Simple and clear.

Approach 2 — Two pointers (optimal): Maintain l, r, lMax, rMax. The pointer on the shorter side limits water capacity. Advance the shorter pointer inward, accumulating water = max - height. O(n) time, O(1) space. The key insight: if lMax < rMax, the water at l is determined solely by lMax (the right wall is guaranteed to be at least rMax > lMax).

Approach 3 — Monotonic stack: Process horizontally, layer by layer. When a taller bar is encountered, calculate water trapped between the popped bar and the new bar. O(n) time, O(n) space. Useful conceptually but the two-pointer approach is preferred in interviews.
