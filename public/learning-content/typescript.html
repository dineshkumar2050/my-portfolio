1. Why TypeScript

TypeScript adds a static type system on top of JavaScript. The compiler catches errors at build time, IDE tooling becomes dramatically more helpful, and code becomes self-documenting through its own type signatures.

JavaScript — runtime error

function greet(user) {
  return "Hello, " + user.naem; // typo
}
// Only fails at runtime
greet({ name: "Alice" });
// → "Hello, undefined"

TypeScript — compile-time error

interface User { name: string }
function greet(user: User): string {
  return "Hello, " + user.naem;
  // Error: Property 'naem' does not exist
}
greet({ name: "Alice" }); // OK

Core Benefits

BenefitWhat It Gives You

TypeScript is a superset: every valid JavaScript program is a valid TypeScript program. Types are erased at compile time — zero runtime overhead.

3. Type Annotations & Inference

TypeScript infers types wherever it has enough context. Explicit annotations add clarity or override inference.

// Inference — let TS figure it out
const count = 5;           // inferred: number
const items = ["a", "b"]; // inferred: string[]
const ratio = count / 2;  // inferred: number

// Annotate when inference is too wide
const ids = [];            // inferred: any[] — bad!
const ids2: number[] = []; // explicit — good

// Function: annotate params, infer return
function add(a: number, b: number) {
  return a + b; // return type inferred as number
}

// Explicit return type — documents intent + catches bugs
function divide(a: number, b: number): number {
  return a / b;
}

// Object destructuring with types
function render({ title, count }: { title: string; count: number }) {
  console.log(title, count);
}
Annotate function parameters always. Annotate return types on public API functions. Let inference work everywhere else.

5. Union & Intersection Types

// Union — A OR B
type StringOrNum = string | number;

function format(val: StringOrNum) {
  if (typeof val === "string") {
    return val.toUpperCase();
  }
  return val.toFixed(2);
}
// Intersection — A AND B
interface Named  { name: string }
interface Aged   { age: number  }
type Person = Named & Aged;

const p: Person = { name: "Ali", age: 25 };
// Must satisfy BOTH interfaces

Discriminated Unions

type Shape =
  | { kind: "circle";    radius: number }
  | { kind: "rectangle"; w: number; h: number }
  | { kind: "triangle";  base: number; height: number };

function area(s: Shape): number {
  switch (s.kind) {
    case "circle":    return Math.PI * s.radius ** 2;
    case "rectangle": return s.w * s.h;
    case "triangle":  return 0.5 * s.base * s.height;
    default:
      const _exhaustive: never = s; // compile error if new kind added
      return _exhaustive;
  }
}
The never exhaustiveness check is the most important pattern with discriminated unions — it breaks the build if you add a new variant without updating the switch.

6. Generics

// Generic function
function identity<T>(x: T): T { return x; }

// Generic with constraint
function getProperty<T, K extends keyof T>(
  obj: T, key: K
): T[K] {
  return obj[key];
}
getProperty({ name: "Bob", age: 20 }, "name"); // string
// Generic class
class Stack<T> {
  private items: T[] = [];
  push(item: T) { this.items.push(item); }
  pop(): T | undefined { return this.items.pop(); }
  peek(): T | undefined {
    return this.items[this.items.length - 1];
  }
}

Generic with Default and Multiple Params

// Default type parameter
type ApiResponse<T = unknown> = {
  data: T;
  status: number;
  message: string;
};

// Multiple type params + constraint
function merge<T extends object, U extends object>(a: T, b: U): T & U {
  return { ...a, ...b };
}
const result = merge({ name: "Ali" }, { age: 30 });
// result: { name: string } & { age: number }

8. Type Guards & Narrowing

// typeof guard
function process(input: string | number) {
  if (typeof input === "string") {
    return input.trim(); // string here
  }
  return input * 2; // number here
}

// instanceof guard
function area(shape: Circle | Rectangle) {
  if (shape instanceof Circle) {
    return Math.PI * shape.radius ** 2;
  }
  return shape.width * shape.height;
}

// in guard
type Cat = { meow(): void };
type Dog2 = { bark(): void };
function speak(pet: Cat | Dog2) {
  if ("meow" in pet) pet.meow();
  else pet.bark();
}

// Custom type predicate
function isString(val: unknown): val is string {
  return typeof val === "string";
}

// Assertion function
function assert(cond: unknown): asserts cond {
  if (!cond) throw new Error("Assertion failed");
}
val is Type predicates narrow the type for the caller — they don't add runtime safety on their own; the body must actually check it.

  constructor(
    public readonly make: string,      // public + readonly
    protected model: string,           // accessible in subclasses
    private _mileage: number = 0       // private

11. Mapped Types, Template Literals & Conditional Types

Mapped Types

// Basic mapped type
type Nullable<T> = { [K in keyof T]: T[K] | null };

// Add readonly and optional modifiers
type DeepPartial<T> = {
  readonly [K in keyof T]?: T[K] extends object
    ? DeepPartial<T[K]>
    : T[K];
};

// Remove readonly with -readonly
type Mutable<T> = { -readonly [K in keyof T]: T[K] };

Template Literal Types

type EventName = "click" | "focus" | "blur";
type Handler = `on${Capitalize<EventName>}`;
// "onClick" | "onFocus" | "onBlur"

type Route = "/users" | "/posts";
type ApiRoute = `GET ${Route}` | `POST ${Route}`;
// "GET /users" | "GET /posts" | ...

Conditional Types & infer

// Conditional type
type IsArray<T> = T extends any[] ? true : false;
type R1 = IsArray<number[]>; // true
type R2 = IsArray<string>;   // false

// infer — extract a type from a structure
type UnpackPromise<T> = T extends Promise<infer U> ? U : T;
type R3 = UnpackPromise<Promise<string>>; // string
type R4 = UnpackPromise<number>;           // number

type FirstArg<F> = F extends (first: infer A, ...rest: any[]) => any ? A : never;

12. Declaration Files (.d.ts)

Declaration files describe the shape of existing JavaScript without providing implementations — essential for consuming JS libraries from TypeScript.

// math-utils.d.ts
declare module "math-utils" {
  export function add(a: number, b: number): number;
  export function clamp(val: number, min: number, max: number): number;
  export const PI: number;
}

// Augment an existing module (e.g. Express Request)
declare module "express" {
  interface Request {
    user?: { id: string; role: string };
  }
}

// Global augmentation
declare global {
  interface Window {
    analytics: { track(event: string): void };
  }
  const __DEV__: boolean;
}
Install community types with npm i -D @types/lodash. Check availability at definitelytyped.org. For packages that include their own types (e.g. axios), no @types install is needed.

// Custom generic hook
function useLocalStorage<T>(key: string, init: T): [T, (v: T) => void] {
  const [val, setVal] = React.useState<T>(() => {
    const stored = localStorage.getItem(key);
    return stored ? JSON.parse(stored) : init;
  });
  const set = (v: T) => { localStorage.setItem(key, JSON.stringify(v)); setVal(v); };
  return [val, set];
}

Q&A — 15 Questions

Easy

Easy What is the difference between any and unknown?

any completely opts out of type checking — you can call any method, index any property, and pass it anywhere without error. It's an escape hatch that silences TypeScript.

unknown is the type-safe counterpart. It also accepts any value, but you cannot perform operations on it until you narrow it to a specific type via a type guard.

let a: any     = "hello";
a.toUpperCase(); // OK — TS doesn't check

let b: unknown = "hello";
// b.toUpperCase();            // Error — must narrow first
if (typeof b === "string") {
  b.toUpperCase(); // OK — narrowed to string
}

Prefer unknown over any whenever you're receiving data from an external source (API, user input) and want TS to enforce safe usage.

Easy What does strictNullChecks do and why does it matter?

Without strictNullChecks, null and undefined are assignable to every type — meaning you can pass null as a string and TypeScript won't complain, leaving the classic billion-dollar mistake uncaught.

With strictNullChecks: true (included in strict), null and undefined are their own types and must be handled explicitly:

function greet(name: string) {
  return "Hello, " + name.toUpperCase();
}
// strictNullChecks OFF: greet(null) compiles, crashes at runtime
// strictNullChecks ON:
greet(null); // Error: Argument of type 'null' is not assignable to 'string'

// Correct signature:
function greet2(name: string | null) {
  if (name === null) return "Hello, stranger";
  return "Hello, " + name.toUpperCase();
}
Easy When should you use a type alias vs an interface?

Use interface when describing the shape of an object that may be extended or implemented by classes, or when you need declaration merging (e.g. augmenting third-party types).

Use type for union types, intersection types, primitive aliases, tuples, and computed types using utility types or conditional types.

The most important rule: be consistent within a codebase. Many teams default to interface for objects and type for everything else.

Easy What is type inference and when does it fail?

TypeScript's type inference automatically assigns types based on assigned values, return statements, and context — you don't always have to write annotations.

Inference fails (becomes too wide or resolves to any) in these common cases:

  • Empty array: const arr = [] → any[]
  • Uninitialized variable: let x; → any
  • Function parameter with no default or annotation
  • Return type of recursive function (sometimes)

Always annotate function parameters and public API surface. Let inference handle local variables and simple return types.

Intermediate

Intermediate Explain structural typing and why it matters.

TypeScript uses structural typing (duck typing): two types are compatible if their shapes match, regardless of their declared names. This is fundamentally different from Java/C# which use nominal typing.

interface Point { x: number; y: number }
interface Vector { x: number; y: number }

function move(p: Point) { return p; }
const v: Vector = { x: 1, y: 2 };
move(v); // OK — structurally compatible even though it's named 'Vector'

class Cat  { meow() {} }
class Dog3 { meow() {} bark() {} }

const c: Cat = new Dog3(); // OK — Dog3 is a structural supertype of Cat

This enables powerful patterns like passing plain objects where interfaces are expected, and it's why you don't need implements on every class.

Intermediate How do discriminated unions enable exhaustiveness checking?

A discriminated union has a common literal property (the "discriminant") on each variant. TypeScript narrows based on this discriminant in switch/if statements.

Exhaustiveness checking uses the never type: after all valid cases are handled, any remaining case must be never. Assigning it to a never-typed variable catches missing cases at compile time — meaning adding a new variant to the union forces you to update every switch over it.

type Action =
  | { type: "INCREMENT" }
  | { type: "DECREMENT" }
  | { type: "RESET"; to: number };

function reducer(state: number, action: Action): number {
  switch (action.type) {
    case "INCREMENT": return state + 1;
    case "DECREMENT": return state - 1;
    case "RESET":     return action.to;
    default:
      const _never: never = action; // add new type → compile error here
      return state;
  }
}
Intermediate What are generic constraints and why use them?

Generic constraints (T extends SomeType) restrict which types a generic can be instantiated with, and also unlock the constrained type's properties within the generic function.

// Without constraint: TS doesn't know T has .length
function getLength<T>(x: T): number {
  return x.length; // Error: Property 'length' does not exist on type 'T'
}

// With constraint: T must have .length
function getLength2<T extends { length: number }>(x: T): number {
  return x.length; // OK
}
getLength2("hello"); // 5
getLength2([1,2,3]); // 3
// getLength2(42);   // Error: number doesn't have .length

// keyof constraint prevents invalid property access
function pluck<T, K extends keyof T>(arr: T[], key: K): T[K][] {
  return arr.map(item => item[key]);
}
Intermediate How does Partial work internally? Implement it yourself.

Partial<T> uses a mapped type that iterates over every key of T and makes the corresponding property optional (?):

// Built-in definition (simplified)
type MyPartial<T> = {
  [K in keyof T]?: T[K];
};

// Usage
interface Config { host: string; port: number; debug: boolean }
type PartialConfig = MyPartial<Config>;
// { host?: string; port?: number; debug?: boolean }

// DeepPartial — recursive variant
type DeepPartial<T> = T extends object
  ? { [K in keyof T]?: DeepPartial<T[K]> }
  : T;

The [K in keyof T] is the mapped type syntax. ? after the key adds optionality. -? removes it (that's how Required works).

Intermediate What is the satisfies operator (TS 4.9) and when should you use it?

satisfies validates that an expression matches a type but preserves the most specific (narrowest) inferred type — unlike a type annotation which widens the type to the annotation.

type Colors = Record<string, [number,number,number] | string>;

// With annotation — too wide, loses specific type info
const c1: Colors = { red: [255,0,0], blue: "#0000ff" };
c1.red; // [number,number,number] | string — can't call .map

// With satisfies — validates shape, keeps specific types
const c2 = {
  red:  [255, 0, 0],
  blue: "#0000ff",
} satisfies Colors;
c2.red.map(x => x * 2); // OK — still [number,number,number]
c2.blue.toUpperCase();   // OK — still string

Use satisfies for configuration objects, palette definitions, route maps — anywhere you want both compile-time validation and narrow inference.

Advanced

Advanced Explain conditional types and the infer keyword with practical examples.

Conditional types (T extends U ? X : Y) choose between types based on whether a type satisfies a condition. infer introduces a type variable to capture part of a matched type.

// Unwrap array element type
type ElementType<T> = T extends (infer E)[] ? E : never;
type E1 = ElementType<string[]>; // string
type E2 = ElementType<number>;   // never

// Extract promise value
type Awaited2<T> = T extends Promise<infer U>
  ? Awaited2<U> // recursive for nested promises
  : T;

// Extract function parameters and return type
type ReturnType2<T extends (...args: any) => any> =
  T extends (...args: any) => infer R ? R : never;

// Distributive conditional types — distributes over union
type Unwrap<T> = T extends Array<infer U> ? U : T;
type R = Unwrap<string[] | number>; // string | number (distributes)
Advanced How do mapped types with template literals enable type-safe event systems?

Combining mapped types with template literal types generates typed event maps from a plain interface definition:

interface AppEvents {
  userLogin:  { userId: string };
  userLogout: { userId: string };
  pageView:   { path: string; duration: number };
}

// Generate on*/off* handler pairs
type EventMap<T> = {
  [K in keyof T as `on${Capitalize<string & K>}`]:
    (payload: T[K]) => void;
};

type Handlers = EventMap<AppEvents>;
// {
//   onUserLogin:  (payload: { userId: string }) => void;
//   onUserLogout: (payload: { userId: string }) => void;
//   onPageView:   (payload: { path: string; duration: number }) => void;
// }

function createEmitter<T>() {
  const handlers: Partial<EventMap<T>> = {};
  return { on: <K extends keyof EventMap<T>>(k: K, fn: EventMap<T>[K]) =>
    { (handlers as any)[k] = fn; }
  };
}
Advanced What is NoInfer<T> (TS 5.4) and what problem does it solve?

When TypeScript infers a type parameter, it considers all inference sites. Sometimes you want a parameter to be inferred from one specific place and validated against that inference elsewhere. NoInfer<T> marks a position as "do not use this for inference".

// Before NoInfer — defaultLang inferred as "en"|"fr"|"de" not caught
function createTranslator<T extends string>(
  langs: T[],
  defaultLang: T  // also infers T, widening it
) { return defaultLang; }
createTranslator(["en","fr"], "de"); // No error! "de" widens T

// With NoInfer — only langs[] drives inference
function createTranslator2<T extends string>(
  langs: T[],
  defaultLang: NoInfer<T> // excluded from inference
) { return defaultLang; }
createTranslator2(["en","fr"], "de"); // Error — "de" not in T
createTranslator2(["en","fr"], "en"); // OK

Very Advanced Implement a type-safe builder pattern using method chaining with generics.

The builder pattern accumulates state across chained calls. Each .set() call adds a new field to the accumulated type, and the .build() method is only callable when all required fields are present.

type Builder<T extends Record<string,unknown> = {}> = {
  set<K extends string, V>(key: K, val: V): Builder<T & Record<K,V>>;
  build(): T;
};

function createBuilder<T extends Record<string,unknown> = {}>(
  data: T = {} as T
): Builder<T> {
  return {
    set(key, val) {
      return createBuilder({ ...data, [key]: val } as any);
    },
    build() { return data; },
  };
}

const user = createBuilder()
  .set("name", "Alice")
  .set("age", 30)
  .set("role", "admin" as const)
  .build();

type Result = typeof user;
// { name: string; age: number; role: "admin" }
user.name; // string
user.role; // "admin"
Very Advanced How would you type a deep-readonly recursive type and handle edge cases like arrays, functions, and Date?

A naive recursive Readonly doesn't handle arrays correctly (they need ReadonlyArray) or leaf types like Date and functions which shouldn't be traversed:

type DeepReadonly<T> =
  T extends (infer U)[]         ? ReadonlyArray<DeepReadonly<U>>
  : T extends Map<infer K, infer V> ? ReadonlyMap<DeepReadonly<K>, DeepReadonly<V>>
  : T extends Set<infer U>     ? ReadonlySet<DeepReadonly<U>>
  : T extends Date | Function  ? T                        // leaf types — don't recurse
  : T extends object           ? { readonly [K in keyof T]: DeepReadonly<T[K]> }
  : T;                                                     // primitive — identity

interface Config {
  db: { host: string; port: number; replicas: string[] };
  created: Date;
}

type FrozenConfig = DeepReadonly<Config>;
// FrozenConfig.db.replicas → ReadonlyArray<string>
// FrozenConfig.created → Date (unchanged)
// Cannot assign to any property — compile error

The key insight is using conditional type ordering: specific structural types (Array, Map, Set) before the generic object branch, and known leaf types before recursing.

Very Advanced Explain how TypeScript's type system can simulate basic arithmetic using recursive conditional types.

TypeScript's type-level computation can represent natural numbers as tuple lengths and implement addition/comparison using recursive types. This is mostly a demonstration of the type system's power, not production code:

// Represent N as a tuple of length N
type BuildTuple<N extends number, T extends unknown[] = []> =
  T["length"] extends N ? T : BuildTuple<N, [...T, unknown]>;

// Add two numbers at the type level
type Add<A extends number, B extends number> =
  [...BuildTuple<A>, ...BuildTuple<B>]["length"];

type Sum = Add<3, 4>; // 7

// Compare: is A less than B?
type LessThan<A extends number, B extends number,
  TA extends unknown[] = BuildTuple<A>,
  TB extends unknown[] = BuildTuple<B>
> = TA["length"] extends TB["length"] ? false
  : TB extends [unknown, ...infer Rest]
    ? TA extends [unknown, ...infer RestA]
      ? LessThan<never, never, RestA, Rest>
      : true
    : false;

type R = LessThan<3, 5>; // true

TypeScript has a recursion depth limit (~1000 levels) so this works for small numbers. The same technique underlies typed routing libraries and SQL query builders that operate entirely at the type level.
