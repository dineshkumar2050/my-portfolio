Angular Learning Reference

1. Overview

Angular is a platform and framework for building single-page client applications using HTML and TypeScript. Developed and maintained by Google, it provides a complete opinionated solution including a powerful CLI, component-based architecture, dependency injection, routing, forms, HTTP client, and testing utilities — all out of the box.

@Component Decorator

import {
  Component, OnInit, OnDestroy, Input, Output, EventEmitter,
  ChangeDetectionStrategy, ViewEncapsulation
} from '@angular/core';

@Component({
  selector: 'app-user-card',
  standalone: true,
  imports: [CommonModule, RouterLink],
  templateUrl: './user-card.component.html',
  styleUrl: './user-card.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
  encapsulation: ViewEncapsulation.Emulated,  // default
  host: {
    'class': 'user-card-host',
    '[attr.data-id]': 'userId'
  }
})
export class UserCardComponent implements OnInit, OnDestroy {
  @Input({ required: true }) userId!: string;
  @Input() role: string = 'viewer';
  @Output() selected = new EventEmitter<string>();

  ngOnInit(): void { /* component ready, inputs set */ }
  ngOnDestroy(): void { /* cleanup subscriptions */ }

  select(): void {
    this.selected.emit(this.userId);
  }
}

Custom Pipe

import { Pipe, PipeTransform } from '@angular/core';

@Pipe({
  name: 'truncate',
  standalone: true,
  pure: true  // default - only runs when input reference changes
})
export class TruncatePipe implements PipeTransform {
  transform(value: string, limit = 50, ellipsis = '...'): string {
    if (!value || value.length <= limit) return value;
    return value.substring(0, limit) + ellipsis;
  }
}

// Usage: {{ longText | truncate:100:'…' }}

Pure vs Impure Pipes

Pure pipe (default): only re-evaluated when the input value reference changes. Very performant. Use for most transformations.

Impure pipe (pure: false): re-evaluated on every change detection cycle. Use when the pipe depends on mutable internal state (e.g., a pipe that filters a mutable array). Can hurt performance — prefer immutable data patterns instead.

8. Services & Dependency Injection

Creating a Service

import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

// providedIn: 'root' — singleton for entire application (tree-shakable)
@Injectable({ providedIn: 'root' })
export class UserService {
  private http = inject(HttpClient);
  private apiUrl = '/api/users';

  getUsers(): Observable<User[]> {
    return this.http.get<User[]>(this.apiUrl);
  }

  getUserById(id: string): Observable<User> {
    return this.http.get<User>(`${this.apiUrl}/${id}`);
  }
}

// providedIn options:
// 'root'     — app-wide singleton (most common)
// 'platform' — shared between multiple Angular apps on the same page
// 'any'      — separate instance per lazy-loaded module
// SomeModule — scoped to that module

Route Guards — Functional (Angular 15+)

import { inject } from '@angular/core';
import { Router, CanActivateFn, CanMatchFn } from '@angular/router';
import { AuthService } from './auth.service';

// Functional guard (preferred in Angular 15+)
export const authGuard: CanActivateFn = (route, state) => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (auth.isLoggedIn()) return true;
  return router.parseUrl('/login?returnUrl=' + state.url);
};

export const roleGuard: CanMatchFn = (route) => {
  const auth = inject(AuthService);
  const requiredRole = route.data?.['role'];
  return auth.hasRole(requiredRole);
};

// CanDeactivate guard
export const unsavedChangesGuard: CanDeactivateFn<FormComponent> =
  (component) => {
    if (component.hasUnsavedChanges()) {
      return confirm('Discard unsaved changes?');
    }
    return true;
  };

// Route with multiple guards
{ path: 'admin', component: AdminComponent,
  canActivate: [authGuard],
  canMatch: [roleGuard],
  canDeactivate: [unsavedChangesGuard] }

Reactive Forms

import { Component, inject } from '@angular/core';
import {
  FormBuilder, FormGroup, FormArray, Validators,
  AbstractControl, ValidationErrors, ReactiveFormsModule
} from '@angular/forms';

@Component({
  selector: 'app-register',
  standalone: true,
  imports: [ReactiveFormsModule],
  templateUrl: './register.component.html'
})
export class RegisterComponent {
  private fb = inject(FormBuilder);

  form = this.fb.group({
    name: ['', [Validators.required, Validators.minLength(2)]],
    email: ['', [Validators.required, Validators.email]],
    password: ['', [Validators.required, Validators.minLength(8)]],
    confirmPassword: [''],
    roles: this.fb.array(['viewer']),
    address: this.fb.group({
      street: ['', Validators.required],
      city:   ['', Validators.required]
    })
  }, { validators: passwordMatchValidator });

  // FormArray helpers
  get roles() { return this.form.get('roles') as FormArray; }
  addRole() { this.roles.push(this.fb.control('')) }
  removeRole(i: number) { this.roles.removeAt(i); }

  // Reactive form events
  ngOnInit() {
    this.form.get('email')?.valueChanges
      .pipe(debounceTime(300), distinctUntilChanged())
      .subscribe(val => this.checkEmail(val));

    this.form.statusChanges.subscribe(status =>
      console.log('Form status:', status));
  }

  onSubmit() {
    if (this.form.valid) {
      console.log(this.form.value);
      // patchValue: update subset of fields
      this.form.patchValue({ name: 'Updated' });
      // setValue: must provide ALL fields
      // this.form.setValue({ name: '...', email: '...' ... });
    } else {
      this.form.markAllAsTouched(); // show all errors
    }
  }
}

// Custom validator
function passwordMatchValidator(g: AbstractControl): ValidationErrors | null {
  const pw = g.get('password')?.value;
  const cpw = g.get('confirmPassword')?.value;
  return pw === cpw ? null : { passwordMismatch: true };
}

// Custom async validator
function uniqueEmailValidator(userSvc: UserService): AsyncValidatorFn {
  return (ctrl: AbstractControl) =>
    userSvc.checkEmailExists(ctrl.value).pipe(
      map(exists => exists ? { emailTaken: true } : null),
      catchError(() => of(null))
    );
}

11. HTTP Client

HttpClient Usage

import { Injectable, inject } from '@angular/core';
import {
  HttpClient, HttpHeaders, HttpParams, HttpErrorResponse
} from '@angular/common/http';
import { Observable, throwError } from 'rxjs';
import { catchError, retry, map } from 'rxjs/operators';

@Injectable({ providedIn: 'root' })
export class ApiService {
  private http = inject(HttpClient);
  private base = 'https://api.example.com';

  getUsers(page: number, role?: string): Observable<User[]> {
    let params = new HttpParams().set('page', page);
    if (role) params = params.set('role', role);

    return this.http.get<User[]>(`${this.base}/users`, { params }).pipe(
      retry(2),
      catchError(this.handleError)
    );
  }

  createUser(user: Partial<User>): Observable<User> {
    const headers = new HttpHeaders({ 'Content-Type': 'application/json' });
    return this.http.post<User>(`${this.base}/users`, user, { headers });
  }

  updateUser(id: string, changes: Partial<User>): Observable<User> {
    return this.http.patch<User>(`${this.base}/users/${id}`, changes);
  }

  deleteUser(id: string): Observable<void> {
    return this.http.delete<void>(`${this.base}/users/${id}`);
  }

  private handleError(err: HttpErrorResponse): Observable<never> {
    let msg = 'An error occurred';
    if (err.error instanceof ErrorEvent) {
      msg = err.error.message; // Client-side
    } else {
      msg = `Server error ${err.status}: ${err.message}`; // Server-side
    }
    console.error(msg);
    return throwError(() => new Error(msg));
  }
}

@Component({ /* ... */ })

    exhaustMap(data => this.userSvc.save(data))

  vm$ = combineLatest({
    users: this.userSvc.getUsers(),
    config: this.userSvc.getConfig(),

  // forkJoin: completes when all observables complete (like Promise.all)
  initData$ = forkJoin({
    users: this.userSvc.getUsers(),
    roles: this.userSvc.getRoles()
  });

  ngOnInit() {
    // takeUntilDestroyed (Angular 16+) — auto-unsubscribes
    this.userSvc.events$.pipe(
      takeUntilDestroyed(this.destroyRef)
    ).subscribe(e => console.log(e));

    // Traditional cleanup pattern
    this.userSvc.events$.pipe(
      takeUntil(this.destroy$)
    ).subscribe(e => console.log(e));
  }

  ngOnDestroy() {
    this.destroy$.next();
    this.destroy$.complete();
  }
}

14. Change Detection

Default vs OnPush

// Default: checks entire component tree on every event/timer/http
@Component({
  changeDetection: ChangeDetectionStrategy.Default
})

// OnPush: only checks when:
// 1. An @Input reference changes
// 2. An event originates from this component or child
// 3. async pipe emits
// 4. Manually triggered (markForCheck/detectChanges)
// 5. Signal read inside template changes
@Component({
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class PureComponent {
  private cdr = inject(ChangeDetectorRef);

  // Force check this component and ancestors
  refresh() { this.cdr.markForCheck(); }

  // Check only this component (not ancestors)
  localCheck() { this.cdr.detectChanges(); }

  // Detach from CD tree entirely (advanced)
  pauseCD() { this.cdr.detach(); }
  resumeCD() { this.cdr.reattach(); }
}

17. Interview Q&A

Easy

Q1: What is the difference between a Component and a Directive in Angular? Easy

A Component is a directive with a template — it has its own view. A Directive adds behavior to existing elements without its own view.

  • Component: has template/templateUrl, creates a host DOM element, used as a custom HTML tag.
  • Structural directive: manipulates the DOM structure (*ngIf, *ngFor). Uses the * microsyntax which desugars to <ng-template>.
  • Attribute directive: changes appearance/behavior of existing elements (NgClass, NgStyle, custom).
// Component — has its own template
@Component({ selector: 'app-card', template: '<div><ng-content></ng-content></div>' })
export class CardComponent {}

// Attribute directive — no template, modifies host element
@Directive({ selector: '[appHighlight]', standalone: true })
export class HighlightDirective {
  @HostBinding('style.background') bg = 'yellow';
}
Q2: What is the purpose of trackBy in *ngFor / @for? Easy

Without trackBy, Angular re-renders ALL items when the array reference changes (e.g., after an API reload). trackBy gives Angular a unique identity for each item so only changed items are re-rendered.

// In component
trackById(index: number, item: User): string { return item.id; }
<!-- Legacy syntax -->
<li *ngFor="let user of users; trackBy: trackById">{{ user.name }}</li>

<!-- New control flow (Angular 17+) — track is REQUIRED -->
@for (user of users; track user.id) {
  <li>{{ user.name }}</li>
}

Performance impact: for a list of 1000 items where only 1 changes, trackBy avoids destroying and recreating 999 DOM nodes.

Q3: What is the async pipe and why is it preferred over manual subscription? Easy

The async pipe subscribes to an Observable or Promise and returns its latest value. It automatically unsubscribes when the component is destroyed, preventing memory leaks.

// Without async pipe — must manually unsubscribe
export class UsersComponent implements OnInit, OnDestroy {
  users: User[] = [];
  private sub!: Subscription;
  ngOnInit() { this.sub = this.userSvc.getUsers().subscribe(u => this.users = u); }
  ngOnDestroy() { this.sub.unsubscribe(); }
}

// With async pipe — automatic subscription + cleanup
export class UsersComponent {
  users$ = this.userSvc.getUsers(); // just expose the observable
}
<!-- Template handles everything -->
@if (users$ | async; as users) {
  @for (user of users; track user.id) {
    <li>{{ user.name }}</li>
  }
}

Benefits: no manual subscription, no memory leaks, works with OnPush change detection (marks view dirty on emission).

Q4: What is the difference between providedIn: 'root' and providing a service in a component's providers array? Easy

providedIn: 'root' registers the service in the application root injector — one singleton instance for the entire app. It is also tree-shakable: if nothing injects it, it is excluded from the bundle.

Providing in a component's providers array creates a new instance scoped to that component and all its descendants. When the component is destroyed, the service instance is also destroyed.

@Injectable({ providedIn: 'root' })
export class GlobalAuthService {} // one singleton

@Component({
  selector: 'app-wizard',
  providers: [WizardStateService]  // new instance per wizard component
})
export class WizardComponent {} // destroyed with component

Use component-level providers for state that should be isolated per instance (e.g., a wizard state service, a form-context service).

Q5: What is ngOnChanges and when does it fire? Easy

ngOnChanges fires whenever a bound @Input property changes. It receives a SimpleChanges object with the previous and current values.

export class UserCardComponent implements OnChanges {
  @Input() userId!: string;

  ngOnChanges(changes: SimpleChanges): void {
    if (changes['userId']) {
      const { previousValue, currentValue, firstChange } = changes['userId'];
      console.log(`userId changed from ${previousValue} to ${currentValue}`);
      if (!firstChange) {
        this.loadUser(currentValue); // reload when input changes
      }
    }
  }
}

Important: it fires before ngOnInit on the first cycle. It only fires for @Input properties — not for internal state changes. With OnPush, it fires on reference changes only (not deep mutations).

Q1: Explain OnPush change detection and how to use it effectively. Intermediate

ChangeDetectionStrategy.OnPush tells Angular to skip change detection for a component unless one of these conditions is met:

  1. An @Input reference changes (shallow comparison)
  2. An event originates from the component or its children
  3. The async pipe receives a new value
  4. markForCheck() or detectChanges() is called
  5. A signal read in the template changes (Angular 17+)
@Component({
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class UserListComponent {
  @Input() users: User[] = [];  // must pass new array reference to trigger

  private cdr = inject(ChangeDetectorRef);
  users$ = this.userSvc.users$; // async pipe handles markForCheck

  // If you need to update manually (e.g., from a callback):
  onExternalUpdate() {
    this.cdr.markForCheck(); // schedule check on next CD cycle
  }
}

// Parent must pass immutable data for OnPush to work correctly
// WRONG: this.users.push(newUser); — same reference, no CD
// RIGHT: this.users = [...this.users, newUser]; — new reference

OnPush is a key performance optimization: it allows Angular to prune large subtrees of the component tree from the CD cycle.

Q2: What are the differences between switchMap, mergeMap, concatMap, and exhaustMap? Intermediate

All four flatten a higher-order Observable (an Observable of Observables) but differ in how they handle concurrent inner observables:

OperatorBehaviorBest Use Case
switchMapCancels previous inner observable, subscribes to latestSearch autocomplete, live filter
mergeMapAll inner observables run concurrentlyParallel requests, fire-and-forget events
concatMapQueues; waits for each inner to complete before nextSequential uploads, ordered processing
exhaustMapIgnores new source emissions while inner is activeForm submit button (prevent double-submit)
// switchMap: typeahead search — cancel stale requests
this.searchInput.valueChanges.pipe(
  debounceTime(300),
  switchMap(term => this.api.search(term))
).subscribe(results => this.results = results);

// exhaustMap: form submit — ignore clicks while submitting
this.submitClicks$.pipe(
  exhaustMap(data => this.api.save(data))
).subscribe(result => this.handleResult(result));

// concatMap: sequential file uploads
from(files).pipe(
  concatMap(file => this.api.upload(file))
).subscribe(res => console.log('Uploaded:', res.name));
Q3: How does the Angular router handle lazy loading, and what are the bundle implications? Intermediate

Lazy loading defers loading a feature's JavaScript chunk until the user navigates to that route. This reduces the initial bundle size and improves startup performance.

// routes.ts — standalone lazy loading (Angular 17+)
export const routes: Routes = [
  {
    path: 'admin',
    loadComponent: () => import('./admin/admin.component')
      .then(m => m.AdminComponent)

  {
    path: 'products',
    loadChildren: () => import('./products/products.routes')

  { path: '', component: ProductListComponent },
  { path: ':id', component: ProductDetailComponent }
];

Q1: Explain Angular's hierarchical dependency injection system and multi-providers. Advanced

Angular maintains a tree of injectors mirroring the component tree. When a token is requested, Angular walks up the injector tree until it finds a provider or throws. This enables scoped services.

// Injector hierarchy (top to bottom):
// Platform injector → Root injector → Module injector → Component injector

// Component-scoped service (new instance per component subtree)
@Component({
  providers: [CartService] // scoped to this component and its children
})
export class CartComponent {
  private cart = inject(CartService); // gets LOCAL instance
}

// Multi-providers: multiple values for one token (additive, not override)
export const HTTP_INTERCEPTORS = new InjectionToken('HTTP_INTERCEPTORS');

providers: [
  { provide: HTTP_INTERCEPTORS, useClass: AuthInterceptor, multi: true },
  { provide: HTTP_INTERCEPTORS, useClass: LoggingInterceptor, multi: true }
]
// inject(HTTP_INTERCEPTORS) returns an array: [AuthInterceptor, LoggingInterceptor]

// @SkipSelf, @Self, @Optional, @Host decorators
@Component({ /* ... */ })
export class ChildComponent {
  // @Self: only look in THIS component's injector (don't go up)
  private svc = inject(MyService, { self: true, optional: true });

  // @SkipSelf: skip this component's injector, start from parent
  private parentSvc = inject(ParentService, { skipSelf: true });

  // @Host: stop looking at the host component (don't cross component boundary)
  private hostSvc = inject(HostService, { host: true });
}
Q2: How do Signals compare to RxJS Observables in Angular, and how do they interoperate? Advanced

Signals and Observables solve different but overlapping problems:

SubscriptionAutomatic (read = track dependency)Manual (subscribe/unsubscribe)

import { toSignal, toObservable } from '@angular/core/rxjs-interop';

@Component({ /* ... */ })
export class UserComponent {
  private userSvc = inject(UserService);

  // Convert Observable → Signal (subscribes automatically, unsubscribes on destroy)
  users = toSignal(this.userSvc.getUsers(), { initialValue: [] });

  searchTerm = signal('');

  // Convert Signal → Observable
  term$ = toObservable(this.searchTerm);
  // term$ emits whenever searchTerm signal changes

Q3: How does Angular SSR (Server-Side Rendering) work in Angular 17+, and what are hydration considerations? Advanced

Angular SSR renders the app on the server (Node.js) and sends fully-formed HTML to the browser. The browser then "hydrates" the static HTML by attaching Angular's event listeners and state without re-rendering from scratch.

// app.config.server.ts

Q4: How would you implement a caching HTTP interceptor with request deduplication? Advanced 
import { HttpInterceptorFn, HttpResponse } from '@angular/common/http';
import { inject } from '@angular/core';
import { Observable, of, Subject } from 'rxjs';
import { tap, share } from 'rxjs/operators';

// Simple in-memory cache service
@Injectable({ providedIn: 'root' })
class HttpCacheService {
  private cache = new Map<string, HttpResponse<unknown>>();
  private inflightRequests = new Map<string, Observable<any>>();

  get(key: string) { return this.cache.get(key); }
  set(key: string, response: HttpResponse<unknown>) {
    this.cache.set(key, response);
    // TTL: auto-evict after 60s
    setTimeout(() => this.cache.delete(key), 60_000);
  }
  getInflight(key: string) { return this.inflightRequests.get(key); }
  setInflight(key: string, obs$: Observable<any>) {
    this.inflightRequests.set(key, obs$);
  }
  deleteInflight(key: string) { this.inflightRequests.delete(key); }
}

export const cachingInterceptor: HttpInterceptorFn = (req, next) => {
  const cache = inject(HttpCacheService);

  // Only cache GET requests with cache-control header
  if (req.method !== 'GET' || !req.headers.has('X-Cache')) {
    return next(req);
  }

  const cacheKey = req.urlWithParams;

  // Return cached response
  const cached = cache.get(cacheKey);
  if (cached) return of(cached.clone());

  // Deduplicate: if same request is in-flight, share it
  const inflight = cache.getInflight(cacheKey);
  if (inflight) return inflight;

  const request$ = next(req).pipe(
    tap(event => {
      if (event instanceof HttpResponse) {
        cache.set(cacheKey, event.clone());
        cache.deleteInflight(cacheKey);
      }
    }),
    share()  // multicast: share one HTTP call to multiple subscribers
  );

  cache.setInflight(cacheKey, request$);
  return request$;
};
Q5: How do you implement a custom structural directive, and explain the desugared template syntax? Advanced 
import {
  Directive, Input, TemplateRef, ViewContainerRef, inject
} from '@angular/core';

// Custom structural directive: *appRepeat="3"
@Directive({ selector: '[appRepeat]', standalone: true })
export class RepeatDirective {
  private vcr = inject(ViewContainerRef);
  private tpl = inject(TemplateRef);

  @Input() set appRepeat(count: number) {
    this.vcr.clear();
    for (let i = 0; i < count; i++) {
      this.vcr.createEmbeddedView(this.tpl, { $implicit: i });
    }
  }
}

// Usage: <li *appRepeat="5; let i">Item {{ i }}</li>

// Angular desugars *appRepeat="5; let i" into:
// <ng-template [appRepeat]="5" let-i>
//   <li>Item {{ i }}</li>
// </ng-template>

The * prefix is syntactic sugar for <ng-template>. Angular:

  1. Wraps the host element in <ng-template>
  2. Moves the *directive="expr" to [directive]="expr" on <ng-template>
  3. Passes let-x variables as template context via $implicit

The directive controls rendering via ViewContainerRef.createEmbeddedView(templateRef, context) and clear().

Q1: Explain the Angular Ivy compilation model — how does the Ivy compiler differ from View Engine? Very Advanced

Ivy (default since Angular 9) replaced the View Engine compiler with a fundamentally different architecture:

// Ivy compiles @Component into static ɵcmp and ɵfac fields:
export class AppComponent {

  static ɵcmp = defineComponent({
    type: AppComponent,
    selectors: [['app-root']],
    decls: 1, vars: 1,
    template: function AppComponent_Template(rf, ctx) {
      if (rf & 1) { text(0); }             // creation phase
      if (rf & 2) { textInterpolate(ctx.title); } // update phase
    }
  });

}

// the NgFor instructions don't ship.

Q2: How does zoneless change detection work, and how do Signals make it possible? Very Advanced

Traditional Angular relies on Zone.js, which monkey-patches async browser APIs (setTimeout, Promise, addEventListener) to notify Angular when async work completes. Angular then runs change detection for the entire app. This is "coarse-grained" — any event anywhere triggers a full tree check.

Signals enable fine-grained, reactive change detection: Angular knows exactly which templates depend on which signals. When a signal changes, only the views that read that signal need to be updated — no zone notification needed.

// Zoneless app setup (Angular 18+, experimental)
export const appConfig: ApplicationConfig = {
  providers: [
    provideExperimentalZonelessChangeDetection(),
    provideRouter(routes)
  ]
};

// Remove zone.js from angular.json polyfills:
// "polyfills": ["zone.js"]  ← REMOVE or keep only for testing

// Component MUST use signals or async pipe for CD to work in zoneless
@Component({
  changeDetection: ChangeDetectionStrategy.OnPush, // required for zoneless
  template: `<p>{{ count() }}</p><button (click)="inc()">+</button>`
})
export class CounterComponent {
  count = signal(0);
  inc() { this.count.update(c => c + 1); }
  // Angular knows count changed → only this template gets re-rendered
}

Zoneless benefits: smaller bundle (no zone.js ~15KB), faster initial render, predictable CD timing, compatible with native async/await and non-Angular code. Libraries that trigger CD outside Angular (e.g., D3 callbacks) need to call ChangeDetectorRef.markForCheck() explicitly.

Q3: Design a scalable state management architecture using Signals without NgRx. Very Advanced 
// Generic signal store factory (Angular 19+ pattern)
import { signal, computed, effect, inject } from '@angular/core';

interface StoreConfig<State, Actions> {
  initialState: State;
  actions: (get: () => State, set: (state: Partial<State>) => void) => Actions;
}

function createStore<State extends object, Actions>(
  config: StoreConfig<State, Actions>
) {
  const state = signal<State>(config.initialState);
  const get = () => state();
  const set = (partial: Partial<State>) =>
    state.update(s => ({ ...s, ...partial }));

  return {
    state: state.asReadonly(),
    actions: config.actions(get, set)
  };
}

// User store
export const useUserStore = createStore({
  initialState: {
    users: [] as User[],
    loading: false,
    error: null as string | null,
    selectedId: null as string | null
  },
  actions: (get, set) => ({
    setUsers(users: User[]) { set({ users, loading: false }); },
    setLoading(loading: boolean) { set({ loading }); },
    setError(error: string) { set({ error, loading: false }); },
    selectUser(id: string) { set({ selectedId: id }); },
    updateUser(updated: User) {
      const users = get().users.map(u => u.id === updated.id ? updated : u);
      set({ users });
    }
  })
});

// Feature service using the store
@Injectable({ providedIn: 'root' })
export class UserFacade {
  private store = useUserStore;
  private http = inject(HttpClient);

  // Expose read-only computed state
  users = computed(() => this.store.state().users);
  loading = computed(() => this.store.state().loading);
  selectedUser = computed(() =>
    this.store.state().users.find(u => u.id === this.store.state().selectedId)
  );

  loadUsers() {
    this.store.actions.setLoading(true);
    this.http.get<User[]>('/api/users').subscribe({
      next: users => this.store.actions.setUsers(users),
      error: err => this.store.actions.setError(err.message)
    });
  }
}

This pattern avoids NgRx boilerplate while maintaining unidirectional data flow, type safety, and fine-grained reactivity. For complex async orchestration or devtools, consider @ngrx/signals SignalStore which formalizes this approach with reducers and effects.

Q4: How do you build a highly reusable form control by implementing ControlValueAccessor? Very Advanced 
import {
  Component, forwardRef, inject, ChangeDetectorRef
} from '@angular/core';
import {
  ControlValueAccessor, NG_VALUE_ACCESSOR,
  NG_VALIDATORS, AbstractControl, ValidationErrors, Validator
} from '@angular/forms';

// Custom form control: star-rating input
@Component({
  selector: 'app-star-rating',
  standalone: true,
  providers: [
    {
      provide: NG_VALUE_ACCESSOR,
      useExisting: forwardRef(() => StarRatingComponent),
      multi: true
    },
    {
      provide: NG_VALIDATORS,
      useExisting: forwardRef(() => StarRatingComponent),
      multi: true
    }
  ],
  template: `
    @for (star of stars; track star) {
      <button type="button"
        [class.filled]="star <= value"
        [disabled]="isDisabled"
        (click)="setValue(star)"
        (blur)="onTouched()">
        ★
      </button>
    }
  `
})
export class StarRatingComponent implements ControlValueAccessor, Validator {
  private cdr = inject(ChangeDetectorRef);
  stars = [1, 2, 3, 4, 5];
  value = 0;
  isDisabled = false;

  // CVA callbacks — set by Angular forms framework
  onChange = (_: number) => {};
  onTouched = () => {};

  // writeValue: called when the form sets the value programmatically
  writeValue(val: number): void {
    this.value = val ?? 0;
    this.cdr.markForCheck();
  }

  // registerOnChange: Angular calls this to give us the "emit upward" fn
  registerOnChange(fn: (val: number) => void): void { this.onChange = fn; }

  // registerOnTouched: Angular calls this to give us the "touched" fn
  registerOnTouched(fn: () => void): void { this.onTouched = fn; }

  // setDisabledState: called when form control disabled state changes
  setDisabledState(disabled: boolean): void {
    this.isDisabled = disabled;
    this.cdr.markForCheck();
  }

  // NG_VALIDATORS
  validate(ctrl: AbstractControl): ValidationErrors | null {
    return this.value > 0 ? null : { starRatingRequired: true };
  }

  setValue(star: number) {
    if (this.isDisabled) return;
    this.value = star;
    this.onChange(star);     // notify form of change
    this.onTouched();        // mark as touched
  }
}

// Usage — works with both reactive and template-driven forms:
// <app-star-rating formControlName="rating"></app-star-rating>
// <app-star-rating [(ngModel)]="productRating"></app-star-rating>
Q5: Explain Angular's runtime compilation model for dynamic components and how to create components at runtime. Very Advanced 
import {
  Component, ViewContainerRef, ComponentRef,
  createComponent, EnvironmentInjector,
  reflectComponentType, inject, Type
} from '@angular/core';

// Approach 1: ViewContainerRef.createComponent (most common)
@Component({ selector: 'app-dynamic-host', template: '<ng-container #container></ng-container>' })
export class DynamicHostComponent {
  @ViewChild('container', { read: ViewContainerRef }) container!: ViewContainerRef;

  loadComponent<T>(componentType: Type<T>, inputs: Record<string, unknown> = {}): ComponentRef<T> {
    this.container.clear();
    const ref = this.container.createComponent(componentType);

    // Pass inputs to the dynamic component
    Object.entries(inputs).forEach(([key, value]) => {
      ref.setInput(key, value);
    });
    ref.changeDetectorRef.detectChanges();
    return ref;
  }

  destroyComponent(ref: ComponentRef<unknown>) { ref.destroy(); }
}

// Approach 2: createComponent (standalone, without ViewContainerRef)
@Injectable({ providedIn: 'root' })
export class DynamicComponentService {
  private injector = inject(EnvironmentInjector);

  create<T>(componentType: Type<T>): ComponentRef<T> {
    return createComponent(componentType, {
      environmentInjector: this.injector
    });
  }
}

// Approach 3: Dynamic component type map (for data-driven UIs)
const WIDGET_REGISTRY = new Map<string, Type<unknown>>([
  ['chart', ChartComponent],
  ['table', TableComponent],
  ['kpi',   KpiComponent]
]);

// Usage in a dashboard builder:
async loadWidget(type: string, container: ViewContainerRef) {
  const component = WIDGET_REGISTRY.get(type);
  if (!component) throw new Error(`Unknown widget: ${type}`);
  container.clear();
  const ref = container.createComponent(component);
  return ref;
}

// Inspect component metadata at runtime
const meta = reflectComponentType(ChartComponent);
console.log(meta?.inputs);   // [{propName, templateName}]
console.log(meta?.outputs);
console.log(meta?.selector);

Dynamic components are the foundation of UI builders, dialog systems, toast notifications, and portal-based UIs. The key shift in Ivy is that components carry their own compilation metadata (ɵcmp), so they can be instantiated without a compile step at runtime.
