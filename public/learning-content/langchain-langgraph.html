LCEL — LangChain Expression Language

LCEL uses the | (pipe) operator to chain components. Every component implements Runnable with invoke(), stream(), batch(), ainvoke(), etc. 
from langchain_core.runnables import (
    RunnableParallel, RunnablePassthrough, RunnableBranch, RunnableLambda
)
from langchain_core.output_parsers import StrOutputParser

llm = ChatAnthropic(model="claude-3-5-sonnet-20241022")
parser = StrOutputParser()

# Basic chain
chain = prompt | llm | parser
result = chain.invoke({"role": "Python", "language": "English", "question": "What is a list?"})

# RunnableParallel — run multiple chains in parallel
parallel_chain = RunnableParallel({
    "answer": prompt | llm | parser,
    "topic": RunnablePassthrough()  # passes input through unchanged
})
result = parallel_chain.invoke({"question": "What is OOP?"})
# result = {"answer": "...", "topic": {"question": "..."}}

# RunnablePassthrough — pass data through while also processing it
rag_chain = (
    {"context": retriever, "question": RunnablePassthrough()}
    | prompt
    | llm
    | parser
)

# RunnableBranch — conditional routing
def classify_question(question: str) -> str:
    if "code" in question.lower(): return "technical"
    if "price" in question.lower(): return "billing"
    return "general"

branch = RunnableBranch(
    (lambda x: classify_question(x["question"]) == "technical", technical_chain),
    (lambda x: classify_question(x["question"]) == "billing", billing_chain),
    general_chain  # default
)

# RunnableLambda — wrap any Python function
def add_timestamp(output: str) -> str:
    from datetime import datetime
    return f"[{datetime.now().isoformat()}] {output}"

chain_with_timestamp = chain | RunnableLambda(add_timestamp)

# Streaming LCEL chain
for chunk in chain.stream({"question": "Explain async/await"}):
    print(chunk, end="", flush=True)

Full RAG Pipeline with LCEL

from langchain_core.runnables import RunnablePassthrough
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_anthropic import ChatAnthropic
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import Chroma
from langchain_community.document_loaders import WebBaseLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter

# --- INDEXING PIPELINE ---
loader = WebBaseLoader("https://docs.langchain.com/docs/concepts")
docs = loader.load()

splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
chunks = splitter.split_documents(docs)

embeddings = OpenAIEmbeddings()
vectorstore = Chroma.from_documents(chunks, embeddings, persist_directory="./rag_db")

# --- RETRIEVAL & GENERATION PIPELINE ---
retriever = vectorstore.as_retriever(search_type="mmr", search_kwargs={"k": 5})
llm = ChatAnthropic(model="claude-3-5-sonnet-20241022")

rag_prompt = ChatPromptTemplate.from_template("""
You are an assistant for question-answering tasks.
Use the following retrieved context to answer the question.
If you don't know the answer, say "I don't know."
Always cite which documents you used.


{context}


Question: {question}
""")

def format_docs(docs):
    return "\n\n".join(f"Source [{i+1}]: {doc.page_content}" for i, doc in enumerate(docs))

rag_chain = (
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
    | rag_prompt
    | llm
    | StrOutputParser()
)

# Invoke
answer = rag_chain.invoke("What is LCEL?")
print(answer)

# Streaming RAG
for chunk in rag_chain.stream("How does memory work in LangChain?"):
    print(chunk, end="", flush=True)

# With citations — return source docs too
from langchain_core.runnables import RunnableParallel
rag_chain_with_sources = RunnableParallel(
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
).assign(answer=rag_prompt | llm | StrOutputParser())
result = rag_chain_with_sources.invoke("What is RAG?")
print(result["answer"])

StateGraph API

from langgraph.graph import StateGraph, END, START
from langchain_anthropic import ChatAnthropic
from langchain_core.messages import HumanMessage, AIMessage
from typing import TypedDict, Annotated
from langgraph.graph.message import add_messages

class State(TypedDict):
    messages: Annotated[list, add_messages]

llm = ChatAnthropic(model="claude-3-5-sonnet-20241022")

def call_llm(state: State) -> dict:
    """Node function: receives state, returns partial update"""
    response = llm.invoke(state["messages"])
    return {"messages": [response]}  # add_messages appends this

def should_continue(state: State) -> str:
    """Conditional edge function: returns next node name"""
    last_message = state["messages"][-1]
    if hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "tools"
    return END

# Build graph
graph = StateGraph(State)

# Add nodes
graph.add_node("llm", call_llm)
graph.add_node("tools", tool_node)

# Set entry point (or use START constant)
graph.set_entry_point("llm")  # equiv: graph.add_edge(START, "llm")

# Add edges
graph.add_edge("tools", "llm")  # after tools, go back to llm

# Conditional edges: (source, router_fn, mapping)
graph.add_conditional_edges(
    "llm",
    should_continue,
    {
        "tools": "tools",  # if router returns "tools", go to "tools" node
        END: END           # if returns END, finish
    }
)

# Compile
app = graph.compile()

# Visualize
print(app.get_graph().draw_mermaid())  # Mermaid diagram
app.get_graph().print_ascii()

Interruptions for Human-in-the-Loop

from langgraph.types import interrupt, Command

# interrupt_before: pause BEFORE specified nodes run
app = graph.compile(
    checkpointer=SqliteSaver.from_conn_string(":memory:"),
    interrupt_before=["tools"],  # Pause before any tool execution
    interrupt_after=["llm"]      # Or pause after specific nodes
)

config = {"configurable": {"thread_id": "hitl-session"}}

# First invoke — runs until interrupt
result = app.invoke({"messages": [HumanMessage("Search for the stock price of AAPL")]}, config)
# Execution paused before "tools" node

# Inspect what's pending
state = app.get_state(config)
print("Pending actions:", state.values["messages"][-1].tool_calls)

# Option 1: Approve — resume execution
result = app.invoke(None, config)  # None = just resume

# Option 2: Reject — update state and resume
app.update_state(config, {"messages": [HumanMessage("Actually, look up GOOGL instead")]})
result = app.invoke(None, config)

# interrupt() function inside a node (dynamic interrupt)
def human_review_node(state):
    decision = interrupt({
        "question": "Should I proceed?",
        "context": state["messages"][-1].content
    })
    return {"approved": decision["approved"]}

# Resume with Command
result = app.invoke(Command(resume={"approved": True}), config)

Streaming Modes

from langchain_core.messages import HumanMessage

config = {"configurable": {"thread_id": "stream-test"}}
input_msg = {"messages": [HumanMessage("Explain Transformers")]}

# stream_mode="values" — full state after each node
print("=== VALUES MODE ===")
for state in app.stream(input_msg, config, stream_mode="values"):
    print(state["messages"][-1].content)

# stream_mode="updates" — partial state returned by each node
print("=== UPDATES MODE ===")
for node_name, update in app.stream(input_msg, config, stream_mode="updates"):
    print(f"Node '{node_name}' returned: {update}")

# stream_mode="messages" — token-by-token LLM output
print("=== MESSAGES MODE (token streaming) ===")
for msg, metadata in app.stream(input_msg, config, stream_mode="messages"):
    if metadata["langgraph_node"] == "llm":  # filter to LLM node
        print(msg.content, end="", flush=True)

# Async streaming
async def stream_async():
    async for chunk in app.astream(input_msg, config, stream_mode="messages"):
        msg, metadata = chunk
        print(msg.content, end="", flush=True)

# --- Worker factory ---
def make_worker(name: str, tools: list, system_prompt: str):
    llm = ChatAnthropic(model="claude-3-5-sonnet-20241022").bind_tools(tools)
    tool_node = ToolNode(tools)

    def worker_fn(state: SupervisorState) -> dict:
        msgs = [SystemMessage(system_prompt)] + state["messages"]
        response = llm.invoke(msgs)
        return {"messages": [AIMessage(content=f"[{name}]: {response.content}")]}

    return worker_fn

research_worker = make_worker("Researcher", [web_search],
    "You are a research specialist. Always search for information before answering.")
coder_worker = make_worker("Coder", [write_code],
    "You are a coding expert. Write clean, well-documented code.")
analyst_worker = make_worker("Analyst", [analyze_data],
    "You are a data analyst. Provide quantitative insights.")

WORKERS = ["Researcher", "Coder", "Analyst", "FINISH"]

def supervisor(state: SupervisorState) -> dict:
    from pydantic import BaseModel

    class Route(BaseModel):
        next: Literal["Researcher", "Coder", "Analyst", "FINISH"]
        reason: str

    llm = ChatAnthropic(model="claude-3-5-sonnet-20241022")

    Route to the best worker, or FINISH if done.
    Current task: {state['messages'][0].content if state['messages'] else ''}"""

    route = structured.invoke([SystemMessage(system)] + state["messages"])
    return {"next_worker": route.next}

# --- Build graph ---

workflow.add_node("supervisor", supervisor)
workflow.add_node("Researcher", research_worker)
workflow.add_node("Coder", coder_worker)

for worker in ["Researcher", "Coder", "Analyst"]:

Interview Questions & Answers

Q1: What is LCEL and why was it introduced? Easy

LCEL (LangChain Expression Language) is a declarative way to compose LangChain components using the | pipe operator. Every component is a Runnable with a consistent interface: invoke(), stream(), batch(), ainvoke().

Why introduced:

  • Eliminates boilerplate from chaining components manually
  • Automatic async, streaming, batch support — write once, works everywhere
  • Automatic LangSmith tracing for every step
  • Parallel execution via RunnableParallel
  • Easy to swap components (change model, parser, retriever)
Q2: Explain the difference between invoke(), stream(), and batch() in LCEL. Easy
  • invoke(input): Runs the chain once synchronously, returns complete output.
  • stream(input): Returns a generator that yields partial outputs as they're generated (token streaming). Essential for chat UI.
  • batch([input1, input2]): Processes multiple inputs in parallel (with configurable max_concurrency). Much faster than sequential invokes.
  • ainvoke/astream/abatch: Async versions for use in async contexts (FastAPI, etc.)

All chains built with LCEL automatically support all these methods, even if the underlying components only implement invoke.

Q3: What is the difference between RecursiveCharacterTextSplitter and SemanticChunker? Easy

RecursiveCharacterTextSplitter: Rule-based — splits at character boundaries (newlines, spaces) with a target chunk_size and overlap. Fast, no LLM needed. Can break semantic units arbitrarily if they exceed chunk_size.

SemanticChunker: Embedding-based — computes sentence embeddings, finds places where semantic similarity drops sharply (topic change), splits there. Preserves semantic coherence. Slower (needs embedding for every sentence). Better for documents with clear topic shifts.

Recommendation: Use RecursiveCharacterTextSplitter as default, switch to SemanticChunker when topic coherence matters more than speed.

Q4: How does MMR (Maximal Marginal Relevance) retrieval differ from simple similarity search? Intermediate

Similarity search: Returns the k most similar documents to the query. Problem: if several chunks come from the same section, they'll all be similar to each other AND the query → redundant results.

MMR: Iteratively selects documents that are: (1) relevant to the query AND (2) different from already selected documents. Formula: λ·sim(doc, query) - (1-λ)·max_sim(doc, selected)

lambda_mult=1.0 = pure similarity, lambda_mult=0.0 = pure diversity. Default ~0.5 balances both.

Best for: when your knowledge base has many similar chunks and you want diverse coverage.

Q5: What are the advantages of EnsembleRetriever combining BM25 and vector search? Intermediate

BM25 (keyword/sparse retrieval): Excellent for exact keyword matches, proper nouns, IDs. Doesn't understand semantics.

Vector search (dense retrieval): Understands semantics and synonyms. Struggles with exact keyword matching and rare terms.

Hybrid (EnsembleRetriever): Combines both using reciprocal rank fusion. Gets both semantic understanding AND precise keyword matching. Consistently outperforms either alone on diverse query types.

Typical weights: 0.3 BM25 + 0.7 semantic, or tune on eval set.

Q7: What is the difference between add_messages reducer and operator.add in LangGraph state? Intermediate

Both append new items to a list, but add_messages is specifically designed for LangChain messages:

  • operator.add: Simple list concatenation: [1,2] + [3] = [1,2,3]
  • add_messages: Also concatenates, but has special handling:
    • If a message with the same id already exists, it UPDATES it (not duplicates)
    • Useful for streaming — same message gets updated chunks
    • Handles ToolMessage pairing with ToolCall IDs

Use add_messages for the messages field, operator.add for other lists like retrieved documents.

Q10: How does LangSmith help debug LangGraph applications? Intermediate

LangSmith provides distributed tracing for every LangGraph execution:

  • Trace view: See every node that executed, inputs/outputs, latency, token counts
  • Nested spans: Tool calls, LLM invocations nested under their parent node
  • Replay: Re-run any trace with the same inputs for debugging
  • Comparison: A/B compare two runs side by side
  • Feedback: Annotate traces with thumbs up/down, use as evaluation dataset
  • Filtering: Search traces by metadata, latency, error type

Enable with LANGCHAIN_TRACING_V2=true. Zero code changes needed for LangGraph — all execution is automatically traced.

from typing import Literal

class AdaptiveRAGState(TypedDict):

    documents: list

Q13: How do you handle very long conversations that exceed the context window in a LangGraph chatbot? Advanced 
from langchain_core.messages import trim_messages, RemoveMessage

def trim_conversation(state: ChatState) -> dict:
    """Keep recent messages within token limit"""
    trimmed = trim_messages(
        state["messages"],
        strategy="last",          # keep last N tokens
        token_counter=llm,        # use model's tokenizer
        max_tokens=4000,
        start_on="human",         # don't cut mid-human message
        end_on=("human", "tool"),
        include_system=True       # always keep system message
    )
    # If messages were trimmed, remove old ones from state
    if len(trimmed) < len(state["messages"]):
        to_remove = [RemoveMessage(id=m.id) for m in state["messages"][:-len(trimmed)]]
        return {"messages": to_remove}
    return {}

# Summarization approach
def summarize_history(state: ChatState) -> dict:
    if len(state["messages"]) > 20:
        summary_prompt = "Summarize this conversation in 3-4 sentences: " + \
            "\n".join(f"{m.type}: {m.content}" for m in state["messages"][:-6])
        summary = llm.invoke([HumanMessage(summary_prompt)]).content
        to_delete = [RemoveMessage(id=m.id) for m in state["messages"][:-6]]
        summary_msg = SystemMessage(f"Previous conversation summary: {summary}")
        return {"messages": to_delete + [summary_msg]}
    return {}
Q14: What is a ContextualCompressionRetriever and when should you use it? Advanced

ContextualCompressionRetriever wraps another retriever and post-processes retrieved documents to extract only the parts relevant to the query.

Problem it solves: Retrieved chunks often contain irrelevant sections — e.g., a chunk retrieved for "Python decorators" might include paragraphs about unrelated topics. This wastes context window space and confuses the LLM.

How it works:

  1. Base retriever fetches top-k documents
  2. Compressor (LLMChainExtractor or EmbeddingsFilter) extracts/scores the relevant parts
  3. Returns compressed, relevant-only content

When to use: When chunks are large and queries are specific. Trade-off: adds LLM call overhead. Use EmbeddingsFilter for faster compression without extra LLM call.

Q15: Explain the PydanticOutputParser vs with_structured_output — when to use each? Intermediate

PydanticOutputParser: Prompt-based — injects format instructions into the prompt asking LLM to output JSON, then parses it. Works with any model. Fragile if LLM doesn't follow format exactly (use OutputFixingParser as fallback). More customizable prompt.

with_structured_output: Uses model's native function calling / tool use / JSON mode. Much more reliable — the model is constrained to output valid JSON matching the schema at the API level. Preferred for newer models (GPT-4, Claude 3+).

Rule of thumb: Use with_structured_output if your model supports it (Claude 3+, GPT-4+). Fall back to PydanticOutputParser for older models or when you need more control over the prompt.

Q16: How does LangGraph's interrupt differ from interrupt_before? Advanced

interrupt_before=["node_name"]: Compile-time configuration — always pauses BEFORE the specified node runs. Good for approval gates you always want. The node hasn't executed when paused.

interrupt(value): Runtime, inside a node's function — pause mid-execution with arbitrary context. The node starts executing but pauses at the interrupt call. Can pass rich context about why pausing.

def review_node(state):
    proposed = generate_proposal(state)
    # Dynamic interrupt with context
    human_response = interrupt({
        "proposed_action": proposed,
        "confidence": 0.7,
        "reason": "High-value action requiring review"
    })
    if human_response["approved"]:
        return execute(proposed)
    return {"rejected": True}

Use interrupt_before for simple pre-node gates. Use interrupt() when the decision to pause depends on runtime data computed inside the node.

Q17: What is MultiQueryRetriever and what problem does it solve? Intermediate

Problem: A single query may not capture all relevant documents. Embedding search is sensitive to phrasing — "ML models" and "AI algorithms" mean similar things but might retrieve different documents.

MultiQueryRetriever: Uses an LLM to generate multiple alternative phrasings of the query (typically 3-5), retrieves documents for each, then unions the results (deduplicating by content).

Example for "What is LangGraph?" might generate: "How does LangGraph work?", "LangGraph state management", "LangGraph vs LangChain agents".

Trade-off: N queries = N embedding searches + 1 LLM call. Best for complex, ambiguous queries where recall matters more than latency.

Q18: How does the RunnableParallel enable performance optimization in RAG? Intermediate 
# Without parallel: sequential — takes sum of all times
result = {
    "docs": retriever.invoke(query),       # 200ms
    "web": search.invoke(query),           # 300ms
    "context": db.invoke(query)            # 150ms
}  # Total: ~650ms

# With RunnableParallel: concurrent — takes max of all times
parallel = RunnableParallel({
    "docs": retriever,
    "web": search,
    "context": db
})
result = parallel.invoke(query)  # Total: ~300ms (max)

RunnableParallel runs all branches concurrently using asyncio.gather under the hood. Each branch is an independent Runnable that can be any chain, retriever, or tool. This is particularly powerful for multi-source retrieval where you want documents from different knowledge bases simultaneously.

Q20: What is a subgraph and when should you use it over a flat graph? Very Advanced

Subgraph: A compiled LangGraph used as a node within a parent graph. Has its own state schema, nodes, edges, and (optionally) checkpointer.

Use subgraphs when:

  • Reusability: The same workflow component is used in multiple parent graphs (authentication, retrieval pipeline)
  • Different state schemas: A sub-workflow needs different state fields than the parent
  • Team boundaries: Different teams own different subgraphs
  • Isolation: Want to test a component independently
  • Complexity: Parent graph is getting too complex — decompose into subgraphs

State compatibility: Parent and subgraph states must have compatible I/O — the parent node receives the subgraph's final state as its output. Use input_schema and output_schema on the subgraph to define the interface.
